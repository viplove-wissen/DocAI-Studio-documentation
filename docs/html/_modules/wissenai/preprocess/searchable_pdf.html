def is_searchable_pdf(file_path):
    """
    Checks if a PDF file is searchable by attempting to extract text.

    Args:
        file_path (str): Path to the PDF file.

    Returns:
        bool: True if the PDF contains searchable text, False otherwise.
    """
    try:
        laparams = LAParams()
        text = extract_text(file_path, laparams=laparams)
        return bool(text.strip())
    except Exception as e:
        logging.error(f"Error checking if PDF is searchable: {e}")
        return False

def perform_ocr(image, dpi=300):
    """
    Performs OCR on an image with optional preprocessing using OpenCV.

    Args:
        image: The OpenCV image object (numpy array).
        dpi: Desired dots-per-inch for the final PIL image (default 300).

    Returns:
        The OCR result as HOCR (HTML).
    """
    try:
        enhanced_image = enhance_image_for_ocr(image)
        pil_image = Image.fromarray(enhanced_image.astype('uint8'))
        resized_image = pil_image.resize((pil_image.width * dpi // 72, pil_image.height * dpi // 72))
        hocr_output = pytesseract.image_to_pdf_or_hocr(resized_image, extension='pdf')
        return hocr_output
    except Exception as e:
        logging.error(f"Error performing OCR: {e}")
        return None

def convert_pdf_to_images(file_path, batch_size= 20, dpi=300):
    """
    Converts a PDF file into a generator that yields batches of page images.

    Args:
        file_path (str): Path to the input PDF file.
        batch_size (int, optional): Number of pages per batch. Defaults to 20.
        dpi (int, optional): Resolution in DPI for image rendering. Defaults to 300.

    Returns:
        generator: Generator yielding lists of PIL Images per batch.
    """
    images = convert_from_path(file_path, dpi=dpi)
    total_pages = len(images)

    def page_generator():
        for start_page in range(1, total_pages + 1, batch_size):
            end_page = min(start_page + batch_size - 1, total_pages)
            yield images[start_page-1:end_page]  # Adjusting index since Python uses 0-based indexing

    return page_generator()

def enhance_dpi(pil_image,new_dpi, old_dpi=150):
    """
    Enhances the resolution (DPI) of a PIL image.

    Args:
        pil_image (PIL.Image): Image to be resized.
        new_dpi (int): Desired DPI.
        old_dpi (int, optional): Original DPI. Defaults to 150.

    Returns:
        PIL.Image: Resized image with enhanced DPI.
    """
    # Calculate the scaling factor based on desired DPI
    old_dpi = int(old_dpi)
    new_dpi = int(new_dpi)
    
    scaling_factor = new_dpi / old_dpi  # Assuming the original DPI is 72
    new_size = (int(pil_image.width * scaling_factor), int(pil_image.height * scaling_factor))
    
    # Resize the image
    resized_image = pil_image.resize(new_size, Image.LANCZOS)

    return resized_image

def enhance_image_for_ocr(image):
    """
    Enhances an image for better OCR performance using grayscale and thresholding.

    Args:
        image (np.ndarray): OpenCV image (NumPy array) in BGR format.

    Returns:
        np.ndarray: Enhanced binary image.
    """
    # Example: Convert to grayscale
    image_np = np.array(image)
    gray_image = cv2.cvtColor(image_np, cv2.COLOR_BGR2GRAY)
    # Example: Apply thresholding
    _, binary_image = cv2.threshold(gray_image, 128, 255, cv2.THRESH_BINARY)
    return binary_image

def extract_images_from_pdf(pdf_path):
    """
    Extracts images from each page of a PDF, applies denoising and DPI enhancement.

    Args:
        pdf_path (str): Path to the input PDF file.

    Returns:
        list: List of tuples (PIL.Image, page_number) for each page.
    """
    # Convert PDF pages into images
    try:
        page_generator = convert_pdf_to_images(pdf_path, batch_size=20, dpi=300)

        total_pages_processed = 0
        images = []

        for pages in page_generator:
            if not pages:
                break
        # pages = convert_from_path(pdf_path, 300)  # You can adjust the resolution (dpi) if needed
            
            for page_num, page_img in enumerate(pages):
                current_page_num = total_pages_processed + page_num + 1
                # Convert PIL image to OpenCV format
                img_array = np.array(page_img)
                # page_height, page_width, _ = img_array.shape

                # Denoise the image using Gaussian blur
                denoised_img = cv2.GaussianBlur(img_array, (5, 5), 0)

                full_image = cv2.resize(denoised_img, None, fx=1, fy=1, interpolation=cv2.INTER_CUBIC)

                # Convert the NumPy arrays to PIL images
                pil_full_image = Image.fromarray(cv2.cvtColor(full_image, cv2.COLOR_BGR2RGB))
                resized_pil_full_image = enhance_dpi(pil_full_image,new_dpi=600,old_dpi=300)
                images.append((resized_pil_full_image, current_page_num))

                # Clear memory for large images
                del full_image,pil_full_image,denoised_img
                gc.collect()
            
            total_pages_processed += len(pages)

        return images
    
    except Exception as e:
        logging.error(f"Error extracting images from PDF: {e}")
        return []

def pdf_to_html(pdf_path, html_path):
    """
    Converts the text content of a searchable PDF into an HTML file.

    Args:
        pdf_path (str): Path to the input PDF file.
        html_path (str): Output path for the HTML file.
    """
    try:
        laparams = LAParams()
        text = extract_text(pdf_path, laparams=laparams)
        
        html_content = "<html>\n<body>\n<pre>\n"
        html_content += html.escape(text)
        html_content += "\n</pre>\n</body>\n</html>"

        with open(html_path, 'w', encoding='utf-8') as f:
            f.write(html_content)
    except Exception as e:
        logging.error(f"Error converting PDF to HTML: {e}")

def make_searchable_pdf(images, file_name, output_directory):
    """
    Creates a searchable PDF by performing OCR on images and combining them.

    Args:
        images (list): List of tuples (PIL.Image, page_number) representing each page.
        file_name (str): Base name for the output PDF.
        output_directory (str): Directory to save the output searchable PDF.

    Returns:
        str: Path to the final searchable PDF file.
    """
    try:
        output_pdf = fitz.open()

        for id, (image, page_number) in enumerate(images):
            logging.info(f"Processing image ID: {id}")
            ocr_result = perform_ocr(image, dpi=300)  # Higher DPI for better quality
            
            if ocr_result:
                try:
                    pdf_stream = fitz.open(stream=ocr_result)
                    output_pdf.insert_pdf(pdf_stream)
                    logging.info(f"OCR result inserted for image ID {id}")
                except Exception as e:
                    logging.error(f"Error inserting OCR result for image ID {id}: {e}")
            else:
                logging.warning(f"OCR result for image ID {id} is missing or empty. Skipping.")
        
        # Save to a temporary file with high-quality settings
        with tempfile.NamedTemporaryFile(delete=False, suffix=".pdf") as tmp_file:
            temp_pdf_path = tmp_file.name
            output_pdf.save(temp_pdf_path, garbage=4, deflate=False, deflate_images=False)
            logging.info(f"Output PDF saved at temporary path >>> {temp_pdf_path}")

        # Optimize and save to the output directory
        output_filename = os.path.join(output_directory, f"{file_name}_searchable.pdf")
        
        # Compress with better settings and quality
        with fitz.open(temp_pdf_path) as pdf:
            pdf.save(output_filename, garbage=4, deflate=False, deflate_images=False)
            logging.info(f"Compressed PDF saved at output path >>> {output_filename}")
        
        return output_filename
        
    except Exception as e:
        logging.error(f"An unexpected error occurred: {e}")
    finally:
        if 'output_pdf' in locals():
            output_pdf.close()
