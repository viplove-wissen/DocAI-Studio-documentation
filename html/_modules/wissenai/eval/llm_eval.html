def eval_rag_auto(query:str, answer:str, contexts: List[List[str]])-> Tuple[float, float]:
    """Evaluate RAG application for two RAGAS metrics that DOES NOT require Ground Truth to be provided.
    These metrics are mathematical and use LLM to make computations by reprompting LLM for specific data.
    Faithfulness: This measures the factual consistency of the generated answer against the given context
    Answer Relevancy: focuses on assessing how pertinent the generated answer is to the given prompt

    Args:
        query (str): Query from user
        answer (str): Final output from the RAG pipeline from the LLM
        contexts (List[List[str]]): top N contexts retrieved from the vector database by retriever

    Returns:
        Tuple[float, float]: A tuple containing two float values:
            - The faithfulness score (float).
            - The answer relevancy score (float).
    """
    # To dict
    data = {
        "question": [query],
        "answer": [answer],
        "contexts": contexts
    }
    # Convert dict to dataset
    dataset = Dataset.from_dict(data)
    result = evaluate_ragas(
    dataset = dataset, 
    metrics=[
        faithfulness,
        answer_relevancy,
    ],
    )
    return result['faithfulness'],result['answer_relevancy']

def eval_rag_groundtruth(query:str, answer:str, contexts: List[List[str]], ground_truth:str)-> Tuple[float, float]:
    """Evaluate RAG applications for two RAGAS metrics that require Ground Truth to be provided.
    These metrics are mathematical and use LLM to make computations by reprompting LLM for specific data.
    Context Precision: metric that evaluates whether all of the ground-truth relevant items present in the contexts are ranked higher or not. 
    Ideally all the relevant chunks must appear at the top ranks.
    Context recall: measures the extent to which the retrieved context aligns with the annotated answer, 
    treated as the ground truth

    Args:
        query (str): Query from user
        answer (str): Final output from the RAG pipeline from the LLM
        contexts (List[List[str]]): top N contexts retrieved from the vector database by retriever
        ground_truth (str): Actual answer to the given question from user

    Returns:
        Tuple[float, float]: A tuple containing two float values:
            - The context_precision score (float).
            - The  context_recall score (float).
    """
    data = {
        "question": [query],
        "answer": [answer],
        "contexts": contexts,
        "ground_truth": [ground_truth]
    }
    # Convert dict to dataset
    dataset = Dataset.from_dict(data)
    result = evaluate_ragas(
    dataset = dataset, 
    metrics=[
        context_precision,
        context_recall,
    ],
    )
    return result['context_precision'],result['context_recall']

def eval_rag_bulk(questions: List[str], answers: List[str], contexts: List[List[List[str]]], ground_truths: List[str]) -> Dict[str, List[float]]:
    """
    Evaluate RAG applications in bulk for two RAGAS metrics that require Ground Truth.
    These metrics are mathematical and use LLM to make computations by reprompting LLM for specific data.
    Includes all of the 4 RAGAS metrics defined in eval_rag_auto and eval_rag_groundtruth

    Args:
        questions (List[str]): List of queries from the user.
        answers (List[str]): List of final outputs from the RAG pipeline from the LLM.
        contexts (List[List[List[str]]]): List of top N contexts retrieved from the vector database by retriever, for each question.
        ground_truths (List[str]): List of actual ground truth answers for the given questions.

    Returns:
        Pandas Dataframe containing, questions, answers and corresponding eval scores
    """
    
    # Create a dictionary of the data for bulk evaluation
    data = {
        "question": questions,
        "answer": answers,
        "contexts": contexts,
        "ground_truth": ground_truths
    }

    # Convert the dictionary into a Dataset
    dataset = Dataset.from_dict(data)

    # Evaluate the dataset using RAGAS metrics

    result = evaluate_ragas(
        dataset=dataset,
        metrics=[
            context_precision,
            context_recall,
            faithfulness,
            answer_relevancy,
        ],
    )
    df = result.to_pandas()
    # Extract the results for context precision and context recall
    return df

def eval_rag_risks_auto(query:str, answer:str, contexts: List[str])-> Tuple[float, float,float]:
    """Evaluate RAG application or LLM outputs in general against 3 DeepEval metrics that Does NOT require
    ground truth to be provided. These metrics are mathematical and use LLM to make computations by
    reprompting LLM for specific data.
    Hallucination: A metric determines whether your LLM generates factually correct information by comparing the actual_output to the provided context
    Bias: A metric determines whether your LLM output contains gender, racial, or political bias
    Toxicity: A metric that evaluates toxicness in your LLM outputs such as Personal attacks, mockery, Hate, Dismissive statements
    Args:
        query (str): Query from user
        answer (str): Final output from the RAG pipeline from the LLM
        contexts (List[str]): top N contexts retrieved from the vector database by retriever

    Returns:
        Tuple[float, float,float]: A tuple containing three float values:
        -Hallucination score
        -Bias score
        -Toxicity score
    """

    test_case = LLMTestCase(
        input=query,
        actual_output=answer,
        context=contexts
    )
    hallucinaion_metric = HallucinationMetric(threshold=0.5,model="gpt-3.5-turbo",include_reason=True)
    hallucinaion_metric.measure(test_case)
    print(f"hallucination score is  {hallucinaion_metric.score}")
    print(hallucinaion_metric.reason)
    
    bias_metric = BiasMetric(threshold=0.5,model="gpt-3.5-turbo",include_reason=True)
    bias_metric.measure(test_case)
    print(f"bias score is {bias_metric.score}")
    print(bias_metric.reason)   
    
    toxic_metric = ToxicityMetric(threshold=0.5,model="gpt-3.5-turbo",include_reason=True)
    toxic_metric.measure(test_case)
    print(f"toxicity score is{toxic_metric.score}")
    print(toxic_metric.reason)
    
    return hallucinaion_metric.score,bias_metric.score,toxic_metric.score

def eval_llm_summary_auto(input:str, llm_output:str)->float:
    """Deep eval metric that evaluates the quality of text summarization performed by LLM

    Args:
        input (str): Actual text to be summarized
        output (str): Summarized output from LLM

    Returns:
        float: Summarization score
    """
    test_case = LLMTestCase(input=input, actual_output=llm_output)
    metric = SummarizationMetric(
        threshold=0.5,
        model="gpt-3.5-turbo"
    )

    metric.measure(test_case)
    #print(metric.score)
    return metric.score

def eval_llm_summary_byword(predictions:List[str],references:List[str])-> Tuple[Dict[str, float], Dict[str, float]]:
    """Evaluation can be done on RAG outputs or LLM summarization outputs using the two scores BLEU and ROUGE.
    These metrics use an algorithm that finds a score which is the word by word difference 
    between the reference (ground truth) and the LLM output.
    BLEU score: measures the similarity between the machine-translated text and the reference translations using n-grams
    ROUGE score measures the similarity between the machine-generated summary and the reference summaries using 
    overlapping n-grams,word sequences that appear in both the machine-generated summary and the reference summaries
    Args:
        predictions (List[str]):  list of predicted sentences
        references (List[str]): list of lists of reference sentences (since each predicted sentence can
        have multiple references), reference sentences refer to ground truth

    Returns:
    Tuple[Dict[str, float], Dict[str, float]]: Two dictionaries containing the BLEU and ROUGE scores, respectively.
    Example output format :
    {'bleu': 1.0, 'precisions': [1.0, 1.0, 1.0, 1.0], 'brevity_penalty': 1.0, 'length_ratio': 1.1666666666666667, 'translation_length': 7, 'reference_length': 6}
    {'rouge1': 1.0, 'rouge2': 1.0, 'rougeL': 1.0, 'rougeLsum': 1.0}
    """
    bleu = evaluate.load("bleu")
    rouge = evaluate.load("rouge")

    # Compute the BLEU score
    results_bleu = bleu.compute(predictions=predictions, references=references)

    # Compute the ROUGE score
    #   original RougeScorer.score has following defn score(self, target, prediction)
    #      target: Text containing the target (ground truth) text,or if a list
    #       prediction: Text containing the predicted text.
    #       hugging face wrapper does score = scorer.score_multi(ref, pred)
    results_rouge = rouge.compute(predictions=predictions, references=references)

    return results_bleu, results_rouge

def eval_llm_summary_byembed(reference:str,candidate:str) -> Dict[str, float]:
    """
    Evaluate the quality of a candidate sentence (model-generated) against a reference sentence (human-provided) 
    BERTScore, a metric that uses embeddings to assess semantic similarity.

    BERTScore calculates precision, recall, and F1 score by comparing embeddings of the candidate and reference sentences.
    It is better than n-gram based metrics 
    Args:
        reference (str): The human-provided reference sentence (ground truth).
        candidate (str): The model-generated candidate sentence to be evaluated.

    Returns:
        Dict[str, float]: A dictionary containing the BERTScore values for Precision, Recall, and F1.
        
        Example return value:
        {
            "Precision": 0.8234,
            "Recall": 0.8123,
            "F1": 0.8178
        }
    """
    scorer = BERTScorer(model_type='bert-base-uncased')
    P, R, F1 = scorer.score([candidate], [reference])
    print(f"BERTScore Precision: {P.mean():.4f}, Recall: {R.mean():.4f}, F1: {F1.mean():.4f}")
    return {
        "Precision": float(f"{P.mean():.4f}"),
        "Recall": float(f"{R.mean():.4f}"),
        "F1": float(f"{F1.mean():.4f}")
    }
