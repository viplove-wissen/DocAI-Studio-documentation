def box_cxcywh_to_xyxy(x):
    x_c, y_c, w, h = x.unbind(-1)
    b = [(x_c - 0.5 * w), (y_c - 0.5 * h), (x_c + 0.5 * w), (y_c + 0.5 * h)]
    return torch.stack(b, dim=1)

def rescale_bboxes(out_bbox, size):
    img_w, img_h = size
    b = box_cxcywh_to_xyxy(out_bbox)
    b = b * torch.tensor([img_w, img_h, img_w, img_h], dtype=torch.float32)
    return b

def outputs_to_objects(outputs, img_size, id2label):
    m = outputs.logits.softmax(-1).max(-1)
    pred_labels = list(m.indices.detach().cpu().numpy())[0]
    pred_scores = list(m.values.detach().cpu().numpy())[0]
    #print(len(pred_labels))
    #print(len(pred_scores))
    pred_bboxes = outputs['pred_boxes'].detach().cpu()[0]
    pred_bboxes = [elem.tolist() for elem in rescale_bboxes(pred_bboxes, img_size)]
    #print(len(pred_bboxes))
    objects = []
    for label, score, bbox in zip(pred_labels, pred_scores, pred_bboxes):
        class_label = id2label[int(label)]
        #print(class_label)
        if not class_label == 'no object':
            objects.append({'label': class_label, 'score': float(score),
                            'bbox': [float(elem) for elem in bbox]})

    return objects

class MaxResize(object):
    def __init__(self, max_size=800):
        self.max_size = max_size

    def __call__(self, image):
        width, height = image.size
        current_max_size = max(width, height)
        scale = self.max_size / current_max_size
        resized_image = image.resize((int(round(scale*width)), int(round(scale*height))))
        return resized_image

class CustomPad(object):
    def __init__(self, padding=(10, 10, 10, 10), padding_value=0):
        self.padding_left, self.padding_top, self.padding_right, self.padding_bottom = padding
        self.padding_value = padding_value

    def __call__(self, image):
        padded_image = transforms.functional.pad(
            image,
            (self.padding_left, self.padding_top, self.padding_right, self.padding_bottom),
            fill=self.padding_value
        )
        return padded_image

class ResizeAndPad(object):
    def __init__(self, max_size=800, padding=(10, 10, 10, 10), padding_value=0):
        self.max_resize = MaxResize(max_size)
        self.custom_pad = CustomPad(padding, padding_value)

    def __call__(self, image):
        resized_image = self.max_resize(image)
        padded_image = self.custom_pad(resized_image)
        return padded_image

def extract_table_struct(image: Image.Image, padding:bool, max_resize:int) -> tuple[Image.Image,list,Image.Image]:
  if padding:
    logging.info("Here to pad")
    pixel_values = struc_pad_transform(image).unsqueeze(0)
  else:
    logging.info("No padding")
    structure_transform = create_structure_transform(max_size=max_resize)
    pixel_values = structure_transform(image).unsqueeze(0)
  pixel_values = pixel_values.to(device)
  logging.info(pixel_values.shape)
  with torch.no_grad():
    outputs = structure_model(pixel_values)
  structure_id2label = structure_model.config.id2label
  structure_id2label[len(structure_id2label)] = "no object"
  logging.info(image.size)
  cells = outputs_to_objects(outputs, image.size, structure_id2label)
  return cells,image

def get_cell_coordinates_by_row(table_data):
    # Extract rows and columns
    rows = [entry for entry in table_data if entry['label'] == 'table row']
    columns = [entry for entry in table_data if entry['label'] == 'table column']

    # Sort rows and columns by their Y and X coordinates, respectively
    rows.sort(key=lambda x: x['bbox'][1])
    columns.sort(key=lambda x: x['bbox'][0])

    # Function to find cell coordinates
    def find_cell_coordinates(row, column):
        cell_bbox = [column['bbox'][0], row['bbox'][1], column['bbox'][2], row['bbox'][3]]
        return cell_bbox

    # Generate cell coordinates and count cells in each row
    cell_coordinates = []

    for row in rows:
        row_cells = []
        for column in columns:
            cell_bbox = find_cell_coordinates(row, column)
            row_cells.append({'column': column['bbox'], 'cell': cell_bbox})

        # Sort cells in the row by X coordinate
        row_cells.sort(key=lambda x: x['column'][0])

        # Append row information to cell_coordinates
        cell_coordinates.append({'row': row['bbox'], 'cells': row_cells, 'cell_count': len(row_cells)})

    # Sort rows from top to bottom
    cell_coordinates.sort(key=lambda x: x['row'][1])

    return cell_coordinates

def apply_ocr(cell_coordinates,orig_img):
    # let's OCR row by row
    data = dict()
    max_num_columns = 0
    for idx, row in enumerate(tqdm(cell_coordinates)):
      row_text = []
      for cell in row["cells"]:
        # crop cell out of image
        cell_image = np.array(orig_img.crop(cell["cell"]))
        # apply OCR
        # result = reader.readtext(np.array(cell_image))
        # if len(result) > 0:
        #   # print([x[1] for x in list(result)])
        #   text = " ".join([x[1] for x in result])
        #   row_text.append(text)
        
        result = ocr.ocr(cell_image, cls=True)
        word = [line[1][0] for res in result if res for line in res]
        text = ' '.join(word)
        
        row_text.append(text)

      if len(row_text) > max_num_columns:
        max_num_columns = len(row_text)

      data[idx] = row_text

    logging.info("Max number of columns:", max_num_columns)

    # pad rows which don't have max_num_columns elements
    # to make sure all rows have the same number of columns
    for row, row_data in data.copy().items():
        if len(row_data) != max_num_columns:
            row_data = row_data + ["" for _ in range(max_num_columns - len(row_data))]
        data[row] = row_data

    return data

def plot_results(cells, class_to_visualize, orig_img, visualize_path):
    if class_to_visualize not in structure_model.config.id2label.values():
        raise ValueError("Class should be one of the available classes")

    # Create a figure and axis but do not show the plot
    fig, ax = plt.subplots(figsize=(16,10))
    ax.imshow(orig_img, interpolation="lanczos")

    for cell in cells:
        score = cell["score"]
        bbox = cell["bbox"]
        label = cell["label"]

        if label == class_to_visualize:
            xmin, ymin, xmax, ymax = tuple(bbox)
            ax.add_patch(plt.Rectangle((xmin, ymin), xmax - xmin, ymax - ymin, fill=False, color="green", linewidth=1))
            text = f'{cell["label"]}: {score:0.2f}'
            ax.text(xmin, ymin, text, fontsize=10, bbox=dict(facecolor='yellow', alpha=0.5))

    plt.axis('off')  # Hide the axis
    # plt.tight_layout()  # Optional: Adjust the layout to avoid clipping

    # Save the figure
    plt.savefig(visualize_path)
    plt.close(fig)

def tsr_inference(input_img_path:str, output_csv_path:str,max_resize:int, visualize: bool, visualize_path:str):
    """Function performs inferencing of images containing tables for table structure recognition using the 
    table transformer for structure recognition model

    Args:
        input_img_path (str): Path to the input image
        output_csv_path (str): Path where the reconstructed output csv file has to be stored
        max_resize(int): Max resize value for the image, Standard value is 800
        visualize (bool): Whether the bounding boxes have to be visualized or not
        visualize_path (str): If bounding boxes have to be visualized, pass the path where the visualized images 
        have to be saved
    """
    file_path = input_img_path
    image = Image.open(file_path).convert("RGB")
    cells, orig_img = extract_table_struct(image, False,max_resize) 
    
    #visualize rows
    if visualize:
        logging.info(f'Going to visualize the tsr extraction for {visualize_path}')
        plot_results(cells, class_to_visualize="table row",orig_img=orig_img,visualize_path=visualize_path)
    
    #get coordinates of each cell
    cell_coordinates = get_cell_coordinates_by_row(cells)
    
    #apply OCR for each cell
    data = apply_ocr(cell_coordinates,orig_img)
    #create csv in specified path
    with open(output_csv_path,'w') as result_file:
        wr = csv.writer(result_file, dialect='excel')
        for row, row_text in data.items():
            wr.writerow(row_text)
    logging.info('Done writing CSV file')
