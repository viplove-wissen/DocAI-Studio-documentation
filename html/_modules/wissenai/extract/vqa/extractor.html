    @staticmethod
    def progress_tracker(total, processed, extractionSessionId):
        """
        Updates the progress percentage for a given document extraction session in the database.

        Args:
            total (int): Total number of items to process.
            processed (int): Number of items processed so far.
            extractionSessionId (str): Identifier for the current extraction session.

        Returns:
            None
        """
        progress = (processed / total) * 100
        update_extraction_progress(docxtractor.db, extractionSessionId, progress)
