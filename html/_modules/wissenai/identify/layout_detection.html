def dict_to_dataframe(data):
    """
    Convert a dictionary with lists of varying lengths into a DataFrame with dynamic column names.

    Parameters:
        data (dict): A dictionary where each key corresponds to a row in the DataFrame, and the value is a list of column values.

    Returns:
        pd.DataFrame: A DataFrame with dynamically generated column names.
    """
    # # Find the maximum length of the lists in the dictionary
    # max_len = max(len(lst) for lst in data.values())
    if data:
        # Use max() with a default value to avoid ValueError
        max_len = max((len(lst) for lst in data.values()), default=0)
    else:
        # Handle the case where data is empty
        max_len = 0

    # Check if data is empty
    if not data:
        print("Warning: Empty data received.")
        return pd.DataFrame()  # Return an empty DataFrame if data is empty

    # Generate dynamic column names based on the maximum length
    column_names = [f'Column{i+1}' for i in range(max_len)]

    # Convert the dictionary to a DataFrame
    df = pd.DataFrame.from_dict(data, orient='index', columns=column_names)

    # Check if the DataFrame has at least one row before setting columns
    if not df.empty and len(df) > 1:
        # Make the first row as column names
        df.columns = df.iloc[0]
        df = df[1:].reset_index(drop=True)
    else:
        print("Warning: DataFrame is too short to reassign column names.")

    return df

def tsr_inference_image(image):
    """Performs table structure recognition (TSR) on an input image.

    Args:
        image (str or PIL.Image): Either a path to an image file or a PIL Image object.

    Returns:
        pd.DataFrame: A DataFrame containing the extracted table data.

    Note:
        The function measures and prints the time taken for table extraction.
    """
    start_time = datetime.now()

    # Assuming `image` is already a PIL image object
    if isinstance(image, str):
        # If the input is a path, convert it to an image
        image = Image.open(image).convert("RGB")
    else:
        image = image.convert("RGB")

    # Extract table structure
    cells, orig_img = extract_table_struct(image, False,max_resize=1000)

    # Get cell coordinates by row
    cell_coordinates = get_cell_coordinates_by_row(cells)

    # Apply OCR to the extracted cells
    data = apply_ocr(cell_coordinates, orig_img)

    # Convert the extracted data to a dataframe
    df = dict_to_dataframe(data)

    end_time = datetime.now()
    print(
        f"Time taken for TTSR for a table extraction ----- start_time {start_time} to end_time {end_time},\n\n duration: {end_time - start_time}")
    print("-" * 100)

    return df

def convert_numpy(data):
    """Recursively converts numpy types to native Python types for JSON serialization.

    Args:
        data: Input data which may contain numpy types (ndarray, integer, floating).

    Returns:
        The converted data with all numpy types replaced by native Python types.
    """
    if isinstance(data, dict):
        return {key: convert_numpy(value) for key, value in data.items()}
    elif isinstance(data, list):
        return [convert_numpy(item) for item in data]
    elif isinstance(data, np.integer):
        return int(data)
    elif isinstance(data, np.floating):
        return float(data)
    elif isinstance(data, np.ndarray):
        return data.tolist()
    elif isinstance(data, pd.DataFrame):
        return data.to_dict(orient='records')  # Convert DataFrame to a list of dictionaries
    else:
        return data

def convert_pdf_to_images(file_path, batch_size=20, dpi=150):
    """Converts a PDF file to a series of images.

    Args:
        file_path (str): Path to the PDF file.
        batch_size (int): Number of pages to process at once.
        dpi (int): DPI (dots per inch) for the output images.

    Returns:
        generator: A generator that yields batches of PIL Image objects.
    """
    images = convert_from_path(file_path, dpi=dpi)
    total_pages = len(images)

    def page_generator():
        for start_page in range(1, total_pages + 1, batch_size):
            end_page = min(start_page + batch_size - 1, total_pages)
            yield images[start_page - 1:end_page]  # Adjusting index since Python uses 0-based indexing

    return page_generator()

def filter_and_sort_headers(data, modified_json_output_filepath):
    """Filters and sorts header blocks (Page-header and Section-header) from layout data.

    Args:
        data (dict): The input layout data.
        modified_json_output_filepath (str): Path to save the modified JSON output.

    Returns:
        tuple: (sorted_data, modified_json_output_filepath) where sorted_data is the filtered and sorted data.
    """
    def sort_blocks_by_min_x(blocks):
        # Sort blocks based on their minimum x-coordinate
        return sorted(blocks, key=lambda block: block['bbox'][0])

    def sort_blocks_by_min_y(blocks):
        # Sort blocks based on their minimum y-coordinate
        return sorted(blocks, key=lambda block: block['bbox'][1])

    def find_headers_and_group(sorted_blocks):
        headers_list = []
        current_group = []
        previous_block = None
        # page_header_id = None

        for i, block in enumerate(sorted_blocks):
            if previous_block:
                prev_xmax = previous_block['bbox'][2]
                prev_xmax_threshold = int(previous_block['bbox'][2])
                if block['bbox'][0] > prev_xmax and block['bbox'][0] > prev_xmax_threshold:
                    if current_group:
                        headers_list.extend(sort_blocks_by_min_y(current_group))
                        current_group = []

            current_group.append(block)
            previous_block = block

        if current_group:
            headers_list.extend(sort_blocks_by_min_y(current_group))

        return headers_list

    result = {}
    for key, blocks in data.items():
        # Filter blocks for Page-header and Section-header labels
        # filtered_blocks = [block for block in blocks if block['label_name'] in ['Page-header', 'Section-header','Table']]
        filtered_blocks = [block for block in blocks]
        sorted_blocks = sort_blocks_by_min_x(filtered_blocks)
        sorted_headers = find_headers_and_group(sorted_blocks)
        result[key] = sorted_headers

    sorted_data = result


    with open(modified_json_output_filepath, 'w') as f:
        json.dump(sorted_data, f, indent=4)

    return sorted_data, modified_json_output_filepath

def layout_detection(img_path, model = model):
    """Performs layout detection on an image using YOLOv10 model.

    Args:
        img_path (str): Path to the input image.
        model: The YOLOv10 model to use for detection.

    Returns:
        tuple: (annotated_image, detections, results) where:
            - annotated_image: The input image with detection annotations
            - detections: The detection results
            - results: Raw model results
    """
    image = cv2.imread(img_path)
    results = model(source=img_path, conf=0.55, iou=0.8)[0]

    detections = sv.Detections.from_ultralytics(results)

    # Custom bounding box color (Red)
    bbox_color = sv.Color(r=255, g=0, b=0)  # RGB format

    # Custom bounding box annotator
    bounding_box_annotator = sv.BoxAnnotator(color=bbox_color)
    label_annotator = sv.LabelAnnotator()

    annotated_image = bounding_box_annotator.annotate(scene=image, detections=detections)
    annotated_image = label_annotator.annotate(scene=annotated_image, detections=detections)

    return annotated_image, detections, results

def enhance_dpi(image, new_dpi=300, old_dpi=150):
    """Enhances the DPI of an image by resizing it.

    Args:
        image (PIL.Image): Input image.
        new_dpi (int): Desired output DPI.
        old_dpi (int): Original DPI of the image.

    Returns:
        PIL.Image: The resized image with enhanced DPI.
    """
    # input is pil image
    # Calculate the scaling factor based on desired DPI
    old_dpi = int(old_dpi)
    new_dpi = int(new_dpi)

    scaling_factor = new_dpi / old_dpi  # Assuming the original DPI is 72
    new_size = (int(image.width * scaling_factor), int(image.height * scaling_factor))

    # Resize the image
    resized_image = image.resize(new_size, Image.LANCZOS)

    return resized_image

def extract_text_from_bbox(image, bbox):
    """Extracts text from a specified bounding box in an image.

    Args:
        image (PIL.Image or np.ndarray): Input image.
        bbox (dict): Bounding box coordinates with keys 'xmin', 'ymin', 'xmax', 'ymax'.

    Returns:
        str: The extracted text from the bounding box region.

    Raises:
        TypeError: If the input image type is not supported.
    """
    # Check image type and convert to OpenCV format if necessary
    if isinstance(image, Image.Image):
        # Convert PIL Image to NumPy array (OpenCV format)
        image = np.array(image)
    elif isinstance(image, np.ndarray):
        # Image is already in OpenCV format, no need to convert
        pass
    else:
        raise TypeError("Unsupported image type. The image should be either a PIL Image or a NumPy array.")

    # Get image dimensions
    image_height, image_width = image.shape[:2]

    # Calculate the bounding box coordinates with padding
    ymin = max(0, bbox['ymin'] - 5)
    ymax = min(image_height, bbox['ymax'] + 5)
    xmin = max(0, bbox['xmin'] - 20)
    xmax = min(image_width, bbox['xmax'] + 20)

    # Crop the image
    cropped_image = image[ymin:ymax, xmin:xmax]

    # Check if the cropped image is empty
    # if cropped_image.size == 0:
    #     return None
    # raise ValueError("The cropped image is empty. Please check the bounding box coordinates.")

    # Convert cropped image to PIL Image
    cropped_image_pil = Image.fromarray(cv2.cvtColor(cropped_image, cv2.COLOR_BGR2RGB))

    # Enhance DPI
    high_dpi_image = enhance_dpi(cropped_image_pil)

    # Convert back to OpenCV format
    high_dpi_image_cv = cv2.cvtColor(np.array(high_dpi_image), cv2.COLOR_RGB2BGR)

    # Convert to grayscale
    gray_image = cv2.cvtColor(high_dpi_image_cv, cv2.COLOR_BGR2GRAY)

    # Extract text
    extracted_text = pytesseract.image_to_string(gray_image)

    return extracted_text

            xmax = min(w, bbox[2] + 10)

            new_bbox = {
                "xmin": int(bbox[0]),
                "ymin": int(bbox[1]),
                "xmax": int(bbox[2]),
                "ymax": int(bbox[3])
            }

            if label_name and detected_label_name == label_name:
                crop_bbox = (new_bbox["xmin"], new_bbox["ymin"], new_bbox["xmax"], new_bbox["ymax"])
                cropped_image = pil_image.crop(crop_bbox)
                cropped_images_path = os.path.join(directory_path,f"{file_name}_cropped_{label_name}")

                os.makedirs(cropped_images_path, exist_ok=True)

                cropped_image_path = os.path.join(
                                                    cropped_images_path,
                                                    f"{label_name}_{idx + 1}.jpeg"
                                                )
                save_cropped_image(cropped_image, cropped_image_path)

                if detected_label_name == 'Table':
                    extracted_df = tsr_inference_image(cropped_image)
                    extracted_text = extracted_df
                    table_id = f"Table_{idx}"
                    save_table_to_excel(extracted_text, table_excel_path,table_id, 1)  # Assuming single page/image input
                else:
                    extracted_text = extract_text_from_bbox(np.array(image), new_bbox)

            elif detected_label_name =='Table':
                crop_bbox = (new_bbox["xmin"], new_bbox["ymin"], new_bbox["xmax"], new_bbox["ymax"])
                cropped_image = pil_image.crop(crop_bbox)
                extracted_df = tsr_inference_image(cropped_image)
                extracted_text = extracted_df
                table_id = f"Table_{idx}"
                save_table_to_excel(extracted_text, table_excel_path,table_id, 1)

            else:
                extracted_text = extract_text_from_bbox(np.array(image), new_bbox)

            page_block_id = f"{str(idx + 1) + '1'}"  # Assuming single page for images
            page_block_id = int(page_block_id)

            page_information.append({
                'page_block_id': page_block_id,
                'label_name': label_name,
                'pdf_page_id': 1,  # Assuming page 1 for images
                'pdf_name': file_name,
                'label_id': class_id,
                'yolo_detection_confidence_score': score,
                'bbox': [xmin, ymin, xmax, ymax],
                'page_img_width': w,
                'page_img_height': h,
                'extracted_text': [extracted_text]
            })

        data_image = {"1": page_information}

        json_output_path = os.path.join(directory_path, f"{file_name}_json_output")
        os.makedirs(json_output_path, exist_ok=True)

        layout_json_file_path = os.path.join(json_output_path, f"yolo_model_detections_{file_name}.json")

        data_image = convert_numpy(data_image)
        layout_list_data = filter_layout_blocks(data_image)

        with open(layout_json_file_path, 'w') as json_file:
            json.dump(data_image, json_file, indent=4)

        sorted_layout_data, sorted_layout_json_filepath = filter_and_sort_layouts(data_image, layout_json_file_path)
        # print("test04")

        end_time = datetime.now()

        print(f"Processed {file_name} from {start_time} to {end_time}, duration: {end_time - start_time}")
        print(f"JSON file created at: {layout_json_file_path}")

        if label_name :
            return sorted_layout_data, class_names,table_excel_dir,bbox_images_path,cropped_images_path

        return sorted_layout_data, class_names, table_excel_dir,bbox_images_path
