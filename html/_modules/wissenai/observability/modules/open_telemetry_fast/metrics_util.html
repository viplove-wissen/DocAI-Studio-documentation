from opentelemetry import metrics

# Obtain a meter instance from the global meter provider
meter = metrics.get_meter_provider().get_meter(__name__)


# Create a counter instrument to track the total number of multilingual API requests
multingual_request_counter = meter.create_counter(
    name="multilingual_requests_total",
    description="Total number of API requests called for multilingual extraction",
    unit="1",
)
