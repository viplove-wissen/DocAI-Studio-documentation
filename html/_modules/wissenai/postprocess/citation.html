class PDFCitationEngine:
    """A class to extract and match fields and values from PDFs, generating citations with bounding boxes.

    Supports both searchable and scanned PDFs using text extraction and OCR engines (Tesseract, PaddleOCR, EasyOCR).
    Matches fields and values using fuzzy matching and spatial analysis, and saves images of matched regions.
    """

    def __init__(self, 
                 max_distance_ratio: float = 0.5,
                 ocr_config: str = '--psm 6',
                 paragraph_threshold: float = 1.2,
                 min_ocr_confidence: int = 60,
                 output_image_dir: str = "output_images",
                 max_edge_distance: float = 150.0,
                 crop_padding: float = 10.0,
                 ocr_engines: List[str] = ["tesseract", "paddleocr", "easyocr"],
                 ocr_priority: List[str] = ["tesseract", "paddleocr", "easyocr"]):
        """Initialize the PDFCitationEngine with configuration parameters.

        Args:
            max_distance_ratio (float): Maximum distance ratio for spatial relationships (currently unused).
            ocr_config (str): Tesseract OCR configuration (e.g., '--psm 6').
            paragraph_threshold (float): Vertical gap threshold for grouping text blocks into paragraphs.
            min_ocr_confidence (int): Minimum confidence score for OCR results (0-100).
            output_image_dir (str): Directory to save output images.
            max_edge_distance (float): Maximum distance for edges in the spatial graph.
            crop_padding (float): Padding around bounding boxes for image cropping.
            ocr_engines (List[str]): List of OCR engines to use ("tesseract", "paddleocr", "easyocr").
            ocr_priority (List[str]): Priority order for OCR engines.

        Attributes:
            field_patterns (defaultdict): Stores alternative patterns for field matching.
            ocr_instances (dict): Initialized OCR engine instances.
        """
        self.max_distance_ratio = max_distance_ratio
        self.ocr_config = ocr_config
        self.paragraph_threshold = paragraph_threshold
        self.min_ocr_confidence = min_ocr_confidence
        self.field_patterns = defaultdict(list)
        self.output_image_dir = output_image_dir
        self.max_edge_distance = max_edge_distance
        self.crop_padding = crop_padding
        self.ocr_engines = ocr_engines
        self.ocr_priority = ocr_priority
        
        # Initialize OCR engines
        self.ocr_instances = {}
        if "tesseract" in ocr_engines:
            self.ocr_instances["tesseract"] = pytesseract
        if "paddleocr" in ocr_engines:
            try:
                self.ocr_instances["paddleocr"] = PaddleOCR(use_angle_cls=True, lang='en')
            except Exception as e:
                print(f"Warning: Failed to initialize PaddleOCR: {str(e)}")
        if "easyocr" in ocr_engines:
            try:
                self.ocr_instances["easyocr"] = easyocr.Reader(['en'], gpu=False)
            except Exception as e:
                print(f"Warning: Failed to initialize EasyOCR: {str(e)}")
        
        # Create output directory
        os.makedirs(self.output_image_dir, exist_ok=True)

    def _preprocess_image(self, image: np.ndarray, engine_name: str) -> np.ndarray:
        """Preprocess an image for OCR with rotation correction and contrast enhancement.

        Args:
            image (np.ndarray): Input image as a NumPy array (RGB).
            engine_name (str): Name of the OCR engine ("tesseract", "paddleocr", "easyocr").

        Returns:
            np.ndarray: Preprocessed image (grayscale or binary).
        """
        gray = cv2.cvtColor(image, cv2.COLOR_RGB2GRAY)
        clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
        contrast = clahe.apply(gray)
        blurred = cv2.GaussianBlur(contrast, (3, 3), 0)
        edges = cv2.Canny(blurred, 50, 150, apertureSize=3)
        lines = cv2.HoughLines(edges, 1, np.pi / 180, 200)
        angle = 0
        if lines is not None:
            angles = []
            for rho, theta in lines[:, 0]:
                angle_rad = theta - np.pi / 2
                angle_deg = angle_rad * 180 / np.pi
                if -45 <= angle_deg <= 45:
                    angles.append(angle_deg)
            if angles:
                angle = np.median(angles)
                print(f"Detected rotation angle: {angle:.2f} degrees")
        
        if abs(angle) > 0.1:
            (h, w) = blurred.shape[:2]
            center = (w // 2, h // 2)
            M = cv2.getRotationMatrix2D(center, angle, 1.0)
            rotated = cv2.warpAffine(blurred, M, (w, h), flags=cv2.INTER_CUBIC, borderMode=cv2.BORDER_REPLICATE)
        else:
            rotated = blurred
        
        if engine_name == "tesseract":
            binary = cv2.adaptiveThreshold(
                rotated, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, 11, 2
            )
            return binary
        return rotated

    def process_pdf(self, pdf_path: str, input_json: List[Dict]) -> List[Dict]:
        """Process a PDF to match fields and values, generating citations with bounding boxes.

        Args:
            pdf_path (str): Path to the PDF file.
            input_json (List[Dict]): JSON structure with companies, fields, and expected values.

        Returns:
            List[Dict]: Results with company names, fields, and citations (bounding boxes, confidence, etc.).

        Raises:
            Exception: If the PDF cannot be opened or processed.
        """
        results = []
        doc = fitz.open(pdf_path)
        all_text_blocks = self._extract_and_group_text_blocks(doc)
        
        for company_entry in input_json:
            company_name = company_entry.get("company", "Unknown_Company")
            company_fields = company_entry.get("fields", company_entry.get("fileds", []))
            
            if not company_fields:
                print(f"Warning: No fields found for company '{company_name}'")
                results.append({
                    "company": company_name,
                    "fields": []
                })
                continue
                
            company_result = {
                "company": company_name,
                "fields": []
            }
            
            for entry in company_fields:
                unique_field_id = entry["unique_filed_id"]
                field_name = next(k for k in entry.keys() if k not in ["unique_filed_id", "page_number"])
                target_value = entry[field_name]
                page_numbers_str = entry["page_number"]
                
                # Initialize field result with all original keys plus citation
                field_result = entry.copy()
                field_result["citation"] = []
                
                page_numbers = []
                try:
                    page_nums = [p.strip() for p in page_numbers_str.split(",") if p.strip()]
                    for p in page_nums:
                        num = int(p)
                        if num > 0:
                            page_numbers.append(num - 1)
                        else:
                            print(f"Warning: Invalid page number '{p}' for {field_name}")
                except ValueError as e:
                    print(f"Warning: Invalid page number format '{page_numbers_str}' for {field_name}: {str(e)}")
                    company_result["fields"].append(field_result)
                    continue
                
                # Track value candidates to retrieve is_paragraph for the highest-confidence citation
                value_candidates_dict = {}
                
                for page_num in page_numbers:
                    if page_num >= len(doc):
                        print(f"Warning: Page {page_num + 1} does not exist")
                        continue
                        
                    page = doc[page_num]
                    page_rect = page.rect
                    page_width = page_rect.width
                    page_height = page_rect.height
                    
                    page_text = page.get_text()
                    page_type = "searchable" if page_text.strip() else "scanned"
                    print(f"Processing page {page_num + 1}: {page_type} PDF")
                    
                    page_blocks = [block for block in all_text_blocks if block['page'] == page_num]
                    if not page_blocks:
                        print(f"Warning: No text blocks found on page {page_num + 1}")
                        continue
                    
                    field_candidates = self._find_field_candidates(field_name, page_blocks)
                    print(f"Field '{field_name}' on page {page_num + 1}: {len(field_candidates)} candidates")
                    
                    value_candidates = self._find_value_candidates(target_value, page_blocks, page_width)
                    print(f"Value '{target_value[:50]}...' on page {page_num + 1}: {len(value_candidates)} candidates")
                    
                    graph = self._build_spatial_graph(page_blocks, page_width, page_height)
                    self._save_graph_image(graph, page_blocks, page_width, page_height, page_num + 1)
                    
                    match = self._find_optimal_pairing_graph(
                        field_candidates, 
                        value_candidates,
                        graph,
                        page_width,
                        page_height
                    )
                    
                    has_complete_match = any(
                        cand.get("is_complete", False) for cand in value_candidates
                    )
                    
                    for value_candidate in value_candidates:
                        is_paragraph = value_candidate.get("is_paragraph", False)
                        partial_bboxes = value_candidate.get("partial_bboxes", [])
                        is_complete = value_candidate.get("is_complete", False)
                        confidence = float(value_candidate.get("fuzzy_score", 0) / 100.0)
                        matched_text = value_candidate.get("text", "")
                        bbox_key = tuple(value_candidate["bbox"])  # Unique key for the bbox
                        value_candidates_dict[bbox_key] = value_candidate  # Store for later reference
                        
                        if has_complete_match and not is_complete:
                            continue
                        
                        if is_paragraph and partial_bboxes and not is_complete:
                            for idx, partial_bbox in enumerate(partial_bboxes):
                                field_result["citation"].append({
                                    "x0": float(partial_bbox["bbox"][0]),
                                    "y0": float(partial_bbox["bbox"][1]),
                                    "x1": float(partial_bbox["bbox"][2]),
                                    "y1": float(partial_bbox["bbox"][3]),
                                    "page_number": str(page_num + 1),
                                    "page_width": float(page_width),
                                    "page_height": float(page_height),
                                    "confidence": confidence,
                                    "matched_text": partial_bbox.get("text", "")
                                })
                        
                        field_result["citation"].append({
                            "x0": float(value_candidate["bbox"][0]),
                            "y0": float(value_candidate["bbox"][1]),
                            "x1": float(value_candidate["bbox"][2]),
                            "y1": float(value_candidate["bbox"][3]),
                            "page_number": str(page_num + 1),
                            "page_width": float(page_width),
                            "page_height": float(page_height),
                            "confidence": confidence,
                            "matched_text": matched_text
                        })
                
                # Sort citations by confidence in descending order
                field_result["citation"] = sorted(field_result["citation"], key=lambda x: x["confidence"], reverse=True)
                
                # Save image for the highest-confidence citation
                if field_result["citation"]:
                    top_citation = field_result["citation"][0]
                    bbox = [top_citation["x0"], top_citation["y0"], top_citation["x1"], top_citation["y1"]]
                    page_num = int(top_citation["page_number"]) - 1
                    safe_field_name = re.sub(r'[^a-zA-Z0-9_]', '_', field_name)
                    # Determine if the top citation is a paragraph by matching its bbox
                    is_paragraph = False
                    for candidate_bbox, candidate in value_candidates_dict.items():
                        if candidate_bbox == tuple(bbox):
                            is_paragraph = candidate.get("is_paragraph", False)
                            break
                    image_type = "complete" if is_paragraph else "single"
                    image_path = self._save_field_image(
                        doc,
                        page_num,
                        bbox,
                        company_name,
                        f"{safe_field_name}_{image_type}_{unique_field_id}",
                        page_num + 1
                    )
                
                company_result["fields"].append(field_result)
            
            results.append(company_result)
        
        doc.close()
        return results



    def _extract_and_group_text_blocks(self, doc: fitz.Document) -> List[Dict]:
        """Extract text blocks from all pages and group them into paragraphs.

        Args:
            doc (fitz.Document): The opened PDF document.

        Returns:
            List[Dict]: List of text blocks with page, text, bounding box, and type information.
        """
        all_blocks = []
        
        for page_num in range(len(doc)):
            page = doc[page_num]
            
            if page.get_text():
                blocks = self._process_searchable_page(page, page_num)
            else:
                blocks = self._process_scanned_page(page, page_num)
            
            grouped_blocks = self._group_into_paragraphs(blocks)
            all_blocks.extend(blocks)
            all_blocks.extend(grouped_blocks)
        
        return all_blocks
    
    def _process_searchable_page(self, page: fitz.Page, page_num: int) -> List[Dict]:
        """Extract text blocks from a searchable PDF page.

        Args:
            page (fitz.Page): The PDF page to process.
            page_num (int): The page number (0-based).

        Returns:
            List[Dict]: List of text blocks with text, bounding box, and metadata.
        """
        blocks = []
        page_blocks = page.get_text("dict")["blocks"]
        
        for b in page_blocks:
            if "lines" in b:
                for line in b["lines"]:
                    for span in line["spans"]:
                        text = span["text"].strip()
                        if text:
                            blocks.append({
                                "page": page_num,
                                "text": text,
                                "bbox": [float(x) for x in span["bbox"]],
                                "type": "text",
                                "block_id": f"b{b['number']}_s{span['size']}_{len(blocks)}"
                            })
        print(f"Page {page_num + 1}: Extracted {len(blocks)} spans")
        return blocks
    
    def _process_scanned_page(self, page: fitz.Page, page_num: int) -> List[Dict]:
        """Extract text blocks from a scanned PDF page using OCR.

        Args:
            page (fitz.Page): The PDF page to process.
            page_num (int): The page number (0-based).

        Returns:
            List[Dict]: List of text blocks with text, bounding box, and OCR metadata.
        """
        blocks = []
        pix = page.get_pixmap()
        img = Image.open(io.BytesIO(pix.tobytes()))
        np_img = np.array(img)
        
        for engine_name in self.ocr_priority:
            if engine_name not in self.ocr_instances:
                print(f"Warning: OCR engine '{engine_name}' not initialized")
                continue
                
            try:
                processed_img = self._preprocess_image(np_img, engine_name)
                
                if engine_name == "tesseract":
                    ocr_data = pytesseract.image_to_data(
                        processed_img, 
                        output_type=pytesseract.Output.DICT,
                        config=self.ocr_config
                    )
                    for i in range(len(ocr_data["text"])):
                        text = ocr_data["text"][i].strip()
                        if text and int(ocr_data["conf"][i]) >= self.min_ocr_confidence:
                            blocks.append({
                                "page": page_num,
                                "text": text,
                                "bbox": [
                                    float(ocr_data["left"][i]),
                                    float(ocr_data["top"][i]),
                                    float(ocr_data["left"][i] + ocr_data["width"][i]),
                                    float(ocr_data["top"][i] + ocr_data["height"][i])
                                ],
                                "type": "ocr_tesseract",
                                "block_id": f"tesseract_word_{i}"
                            })
                
                elif engine_name == "paddleocr":
                    result = self.ocr_instances["paddleocr"].ocr(processed_img, cls=True)
                    if result:
                        for line in result[0]:
                            bbox = line[0]
                            text = line[1][0].strip()
                            confidence = line[1][1]
                            if text and confidence >= self.min_ocr_confidence / 100:
                                blocks.append({
                                    "page": page_num,
                                    "text": text,
                                    "bbox": [
                                        float(bbox[0][0]),
                                        float(bbox[0][1]),
                                        float(bbox[2][0]),
                                        float(bbox[2][1])
                                    ],
                                    "type": "ocr_paddleocr",
                                    "block_id": f"paddleocr_{len(blocks)}"
                                })
                
                elif engine_name == "easyocr":
                    result = self.ocr_instances["easyocr"].readtext(processed_img, detail=1)
                    for detection in result:
                        bbox = detection[0]
                        text = detection[1].strip()
                        confidence = detection[2]
                        if text and confidence >= self.min_ocr_confidence / 100:
                            blocks.append({
                                "page": page_num,
                                "text": text,
                                "bbox": [
                                    float(bbox[0][0]),
                                    float(bbox[0][1]),
                                    float(bbox[2][0]),
                                    float(bbox[2][1])
                                ],
                                "type": "ocr_easyocr",
                                "block_id": f"easyocr_{len(blocks)}"
                            })
                
                if blocks:
                    print(f"Page {page_num + 1}: Extracted {len(blocks)} blocks with {engine_name}")
                    break
                    
            except Exception as e:
                print(f"Error with {engine_name} on page {page_num + 1}: {str(e)}")
                continue
        
        if not blocks:
            print(f"Warning: No text blocks extracted for page {page_num + 1} with any OCR engine")
        
        return blocks
    
    def _group_into_paragraphs(self, blocks: List[Dict]) -> List[Dict]:
        """Group text blocks into paragraphs based on spatial proximity.

        Args:
            blocks (List[Dict]): List of text blocks to group.

        Returns:
            List[Dict]: List of grouped paragraph blocks.
        """
        if not blocks:
            return []
        
        sorted_blocks = sorted(blocks, key=lambda x: (x["bbox"][1], x["bbox"][0]))
        paragraphs = []
        current_para = []
        
        for i, block in enumerate(sorted_blocks):
            if not current_para:
                current_para.append(block)
                continue
                
            prev_block = current_para[-1]
            vertical_gap = block["bbox"][1] - prev_block["bbox"][3]
            avg_line_height = np.mean([b["bbox"][3] - b["bbox"][1] for b in current_para])
            horizontal_diff = abs(block["bbox"][0] - prev_block["bbox"][0])
            
            if (vertical_gap < avg_line_height * self.paragraph_threshold and
                horizontal_diff < 20):
                current_para.append(block)
            else:
                paragraphs.append(self._create_paragraph_block(current_para))
                current_para = [block]
        
        if current_para:
            paragraphs.append(self._create_paragraph_block(current_para))
        
        return paragraphs
    
    def _create_paragraph_block(self, blocks: List[Dict]) -> Dict:
        """Create a paragraph block from a list of text blocks.

        Args:
            blocks (List[Dict]): List of text blocks to combine.

        Returns:
            Dict: A paragraph block with combined text and bounding box.
        """
        combined_text = ' '.join([b["text"] for b in blocks])
        x0 = float(min(b["bbox"][0] for b in blocks))
        y0 = float(min(b["bbox"][1] for b in blocks))
        x1 = float(max(b["bbox"][2] for b in blocks))
        y1 = float(max(b["bbox"][3] for b in blocks))
        
        return {
            "page": blocks[0]["page"],
            "text": combined_text,
            "bbox": [x0, y0, x1, y1],
            "type": blocks[0]["type"],
            "is_paragraph": True,
            "component_blocks": blocks,
            "block_id": f"para_{blocks[0]['block_id']}"
        }
    
    def _find_field_candidates(self, field_name: str, blocks: List[Dict]) -> List[Dict]:
        """Find text blocks that match a field name or its alternative patterns.

        Args:
            field_name (str): The field name to search for.
            blocks (List[Dict]): List of text blocks to search.

        Returns:
            List[Dict]: List of matching field candidates with text and bounding boxes.
        """
        candidates = []
        patterns = [re.escape(field_name)] + self.field_patterns.get(field_name, [])
        
        for pattern in patterns:
            for block in blocks:
                if re.search(pattern, block["text"], re.IGNORECASE):
                    candidates.append({
                        "text": block["text"],
                        "bbox": [float(x) for x in block["bbox"]],
                        "type": block["type"],
                        "pattern": pattern,
                        "block_id": block.get("block_id", "")
                    })
        
        return candidates
    
    def _find_value_candidates(self, target_value: str, blocks: List[Dict], page_width: float) -> List[Dict]:
        """Find text blocks that match a target value, including paragraph matches.

        Args:
            target_value (str): The value to search for.
            blocks (List[Dict]): List of text blocks to search.
            page_width (float): Width of the page for spatial calculations.

        Returns:
            List[Dict]: List of matching value candidates with text, bounding boxes, and metadata.
        """
        is_paragraph = len(target_value.split()) > 10
        candidates = []
        exact_candidates = self._find_exact_value_candidates(target_value, blocks)
        print(f"Exact candidates for '{target_value[:50]}...': {len(exact_candidates)}")
        candidates.extend(exact_candidates)
        
        if is_paragraph:
            para_candidates = self._find_paragraph_candidates(target_value, blocks, page_width, is_paragraph)
            print(f"Paragraph candidates for '{target_value[:50]}...': {len(para_candidates)}")
            candidates.extend(para_candidates)
        
        seen = set()
        unique_candidates = []
        for cand in candidates:
            key = (tuple(cand["bbox"]), cand["text"])
            if key not in seen:
                seen.add(key)
                unique_candidates.append(cand)
        
        return unique_candidates
    
    def _find_exact_value_candidates(self, target_value: str, blocks: List[Dict]) -> List[Dict]:
        """Find text blocks with exact or fuzzy matches for a target value.

        Args:
            target_value (str): The value to search for.
            blocks (List[Dict]): List of text blocks to search.

        Returns:
            List[Dict]: List of matching value candidates with text, bounding boxes, and fuzzy scores.
        """
        candidates = []
        target_clean = re.sub(r'\s+', ' ', target_value.strip()).lower()
        
        for block in blocks:
            block_text_clean = re.sub(r'\s+', ' ', block["text"].strip()).lower()
            fuzzy_score = fuzz.ratio(target_clean, block_text_clean)
            if block_text_clean.strip() == target_clean.strip() or fuzzy_score > 50:
                candidates.append({
                    "text": block["text"],
                    "bbox": [float(x) for x in block["bbox"]],
                    "type": block["type"],
                    "is_paragraph": block.get("is_paragraph", False),
                    "is_complete": True,
                    "match_type": "exact" if block_text_clean.strip() == target_clean.strip() else "fuzzy",
                    "block_id": block.get("block_id", ""),
                    "fuzzy_score": fuzzy_score
                })
                print(f"Match found for '{target_value[:50]}...' at bbox {block['bbox']}: {'exact' if block_text_clean.strip() == target_clean.strip() else f'fuzzy (score={fuzzy_score})'}")
        
        return candidates
    
    def _find_paragraph_candidates(self, target_value: str, blocks: List[Dict], page_width: float, is_paragraph: bool) -> List[Dict]:
        """Find paragraph candidates by matching sequences of text blocks.

        Args:
            target_value (str): The value to search for.
            blocks (List[Dict]): List of text blocks to search.
            page_width (float): Width of the page for spatial calculations.
            is_paragraph (bool): Whether the target value is expected to be a paragraph.

        Returns:
            List[Dict]: List of paragraph candidates with combined text and bounding boxes.
        """
        candidates = []
        target_clean = re.sub(r'\s+', ' ', target_value.strip()).lower()
        target_words = target_clean.split()
        
        if len(target_words) <= 2 and bool(re.match(r'^-?\d*\.?\d+$', target_value)):
            return []
        
        span_blocks = [b for b in blocks if not b.get("is_paragraph", False)]
        sorted_blocks = sorted(span_blocks, key=lambda x: (x["bbox"][1], x["bbox"][0]))
        
        i = 0
        while i < len(sorted_blocks):
            current_blocks = []
            current_text = []
            current_pos = i
            target_pos = 0
            partial_bboxes = []
            fuzzy_scores = []
            
            while current_pos < len(sorted_blocks) and target_pos < len(target_words):
                block_text = re.sub(r'\s+', ' ', sorted_blocks[current_pos]["text"].strip()).lower()
                block_words = block_text.split()
                fuzzy_score = fuzz.partial_ratio(' '.join(block_words), ' '.join(target_words[target_pos:target_pos + len(block_words)]))
                
                if fuzzy_score > 50:
                    current_blocks.append(sorted_blocks[current_pos])
                    current_text.append(block_text)
                    partial_bboxes.append({
                        "bbox": [float(x) for x in sorted_blocks[current_pos]["bbox"]],
                        "text": sorted_blocks[current_pos]["text"]
                    })
                    fuzzy_scores.append(fuzzy_score)
                    target_pos += len(block_words)
                    current_pos += 1
                    
                    if current_blocks and len(current_blocks) > 1:
                        prev_block = current_blocks[-2]
                        curr_block = current_blocks[-1]
                        vertical_gap = curr_block["bbox"][1] - prev_block["bbox"][3]
                        horizontal_diff = abs(curr_block["bbox"][0] - prev_block["bbox"][0])
                        avg_line_height = np.mean([b["bbox"][3] - b["bbox"][1] for b in current_blocks])
                        if vertical_gap > avg_line_height * self.paragraph_threshold or horizontal_diff > 20:
                            break
                else:
                    break
            
            combined_text = ' '.join(current_text).strip()
            overall_fuzzy_score = fuzz.ratio(combined_text, target_clean) if combined_text else 0
            if combined_text and (combined_text == target_clean or overall_fuzzy_score > 50):
                x0 = float(min(b["bbox"][0] for b in partial_bboxes))
                y0 = float(min(b["bbox"][1] for b in partial_bboxes))
                x1 = float(max(b["bbox"][2] for b in partial_bboxes))
                y1 = float(max(b["bbox"][3] for b in partial_bboxes))
                
                candidates.append({
                    "text": combined_text,
                    "bbox": [x0, y0, x1, y1],
                    "type": current_blocks[0]["type"],
                    "is_paragraph": True,
                    "is_complete": True,
                    "match_type": "exact_sequence" if combined_text == target_clean else "fuzzy_sequence",
                    "block_id": f"seq_{current_blocks[0]['block_id']}",
                    "partial_bboxes": partial_bboxes,
                    "fuzzy_score": overall_fuzzy_score
                })
                print(f"Found paragraph match for '{target_value[:50]}...': {'exact' if combined_text == target_clean else f'fuzzy (score={overall_fuzzy_score})'} at bbox [{x0}, {y0}, {x1}, {y1}]")
            elif combined_text and overall_fuzzy_score > 50:
                x0 = float(min(b["bbox"][0] for b in partial_bboxes))
                y0 = float(min(b["bbox"][1] for b in partial_bboxes))
                x1 = float(max(b["bbox"][2] for b in partial_bboxes))
                y1 = float(max(b["bbox"][3] for b in partial_bboxes))
                
                candidates.append({
                    "text": combined_text,
                    "bbox": [x0, y0, x1, y1],
                    "type": current_blocks[0]["type"],
                    "is_paragraph": True,
                    "is_complete": False,
                    "match_type": "partial_fuzzy",
                    "block_id": f"partial_seq_{current_blocks[0]['block_id']}",
                    "partial_bboxes": partial_bboxes,
                    "fuzzy_score": overall_fuzzy_score
                })
                print(f"Found partial paragraph match for '{target_value[:50]}...': fuzzy (score={overall_fuzzy_score}) at bbox [{x0}, {y0}, {x1}, {y1}]")
            
            i = current_pos if current_pos > i else i + 1
        
        return candidates
    
    def _build_spatial_graph(self, blocks: List[Dict], page_width: float, page_height: float) -> Dict[str, List[Tuple[str, float]]]:
        """Build a spatial graph of text blocks based on their proximity.

        Args:
            blocks (List[Dict]): List of text blocks.
            page_width (float): Width of the page.
            page_height (float): Height of the page.

        Returns:
            Dict[str, List[Tuple[str, float]]]: Graph with block IDs as nodes and distances as edge weights.
        """
        graph = defaultdict(list)
        
        for i, block1 in enumerate(blocks):
            center1 = self._get_bbox_center(block1["bbox"])
            for j, block2 in enumerate(blocks[i+1:], start=i+1):
                center2 = self._get_bbox_center(block2["bbox"])
                dist = float(distance.euclidean(center1, center2))
                
                if dist <= self.max_edge_distance:
                    graph[block1["block_id"]].append((block2["block_id"], dist))
                    graph[block2["block_id"]].append((block1["block_id"], dist))
        
        return graph

    def _dijkstra(self, graph: Dict[str, List[Tuple[str, float]]], start: str) -> Dict[str, float]:
        """Run Dijkstra's algorithm to find shortest paths from a starting node.

        Args:
            graph (Dict[str, List[Tuple[str, float]]]): The spatial graph.
            start (str): Starting node ID.

        Returns:
            Dict[str, float]: Shortest distances from the start node to all other nodes.
        """
        distances = {node: float('inf') for node in graph}
        distances[start] = 0
        pq = [(0, start)]
        
        while pq:
            current_dist, current_node = heapq.heappop(pq)
            
            if current_dist > distances[current_node]:
                continue
                
            for neighbor, weight in graph[current_node]:
                distance_to_neighbor = current_dist + float(weight)
                
                if distance_to_neighbor < distances[neighbor]:
                    distances[neighbor] = distance_to_neighbor
                    heapq.heappush(pq, (distance_to_neighbor, neighbor))
        
        return distances
    
    def _find_optimal_pairing_graph(self, 
                                  field_candidates: List[Dict], 
                                  value_candidates: List[Dict],
                                  graph: Dict[str, List[Tuple[str, float]]],
                                  page_width: float,
                                  page_height: float) -> Optional[Dict]:
        """Find the optimal field-value pair based on spatial proximity.

        Args:
            field_candidates (List[Dict]): List of field candidates.
            value_candidates (List[Dict]): List of value candidates.
            graph (Dict[str, List[Tuple[str, float]]]): The spatial graph.
            page_width (float): Width of the page.
            page_height (float): Height of the page.

        Returns:
            Optional[Dict]: Best matching pair with field and value details, or None if no match.
        """
        if not field_candidates or not value_candidates:
            return None
        
        best_match = None
        min_score = float('inf')
        
        for field in field_candidates:
            field_distances = self._dijkstra(graph, field["block_id"])
            
            for value in value_candidates:
                value_distances = self._dijkstra(graph, value["block_id"])
                
                common_nodes = set(field_distances.keys()) & set(value_distances.keys())
                if not common_nodes:
                    continue
                
                score = 0
                for node in common_nodes:
                    dist_to_field = field_distances[node]
                    dist_to_value = value_distances[node]
                    score += (dist_to_field + dist_to_value) / 2
                
                score /= len(common_nodes)
                
                if value.get("is_paragraph", False) and len(value["text"].split()) > 10:
                    score *= 0.8
                
                normalized_score = score / min(page_width, page_height)
                
                if normalized_score < min_score:
                    min_score = normalized_score
                    best_match = {
                        "field_bbox": [float(x) for x in field["bbox"]],
                        "value_bbox": [float(x) for x in value["bbox"]],
                        "confidence": float(1 - min(normalized_score, 1.0)),
                        "is_paragraph": value.get("is_paragraph", False),
                        "matched_text": value["text"],
                        "field_text": field["text"]
                    }
        
        return best_match
    
    def _get_bbox_center(self, bbox: List[float]) -> Tuple[float, float]:
        """Calculate the center of a bounding box.

        Args:
            bbox (List[float]): Bounding box coordinates [x0, y0, x1, y1].

        Returns:
            Tuple[float, float]: Center coordinates (x, y).
        """
        return (float((bbox[0] + bbox[2]) / 2), float((bbox[1] + bbox[3]) / 2))

    def _save_field_image(self, doc: fitz.Document, page_num: int, bbox: List[float], 
                         company_name: str, field_name: str, display_page_num: int) -> str:
        """Save an image of a bounding box region from a PDF page.

    def _convert_numpy_types(self, obj: Any) -> Any:
        """Recursively convert NumPy types to Python native types for JSON serialization.

        Args:
            obj (Any): Object to convert (dict, list, NumPy type, etc.).

        Returns:
            Any: Object with NumPy types converted to Python types.
        """
        if isinstance(obj, np.integer):
            return int(obj)
        elif isinstance(obj, np.floating):
            return float(obj)
        elif isinstance(obj, np.ndarray):
            return obj.tolist()
        elif isinstance(obj, dict):
            return {key: self._convert_numpy_types(value) for key, value in obj.items()}
        elif isinstance(obj, list):
            return [self._convert_numpy_types(item) for item in obj]
        elif isinstance(obj, tuple):
            return tuple(self._convert_numpy_types(item) for item in obj)
        return obj
