def check_and_extract_interest(text):
    """Extracts interest or discounting charge percentage from text using DistilBERT.

    Checks if the input text contains keywords related to interest or discounting charges.
    If found, uses a DistilBERT question-answering model to extract the percentage value.

    Args:
        text (str): The input text to analyze for interest-related keywords and values.

    Returns:
        str or None: The extracted interest percentage (e.g., '8.50%') if found, otherwise None.
    """
    keywords = ['Interest', 'Discounting Charges', 'Interest / Discounting Charges', 
                ' Interest / Discounting Charges', 'INTEREST DISCOUNTING CHARGES', 
                'INTEREST / DISCOUNTING CHARGES', '2ZINTEREST OR DISCOUNTING CHARGES']
    for keyword in keywords:
        if keyword.lower() in text.lower():
            tokenizer = DistilBertTokenizer.from_pretrained('distilbert-base-uncased-distilled-squad')
            model = DistilBertForQuestionAnswering.from_pretrained('distilbert-base-uncased-distilled-squad')

            question = "What is the interest discounting charge in percentage?" 

            inputs = tokenizer(question, text, return_tensors="pt")
            with torch.no_grad():
                outputs = model(**inputs)

            answer_start_index = torch.argmax(outputs.start_logits)
            answer_end_index = torch.argmax(outputs.end_logits)

            predict_answer_tokens = inputs.input_ids[0, answer_start_index : answer_end_index + 1]
            answer = tokenizer.decode(predict_answer_tokens)
            print("Answer of the Question is :", tokenizer.decode(predict_answer_tokens))
            return answer
        else:
            None

def postprocess_interest_value(text):
    """Cleans an extracted interest value by removing non-numeric characters.

    Removes all characters except digits, spaces, dots, and hyphens from the input text
    to extract a clean numerical value.

    Args:
        text (str): The text containing the interest value (e.g., '8.50%').

    Returns:
        str or None: The cleaned numerical value (e.g., '8.50') if non-empty, otherwise None.
    """
    cleaned_text = re.sub(r'[^\d\s.-]', '', text)
    print("CLeaned Text for Interest \n", cleaned_text)
    if cleaned_text:
        return cleaned_text
    else:
        None

def process_interest_values(input_df):
    """Processes a DataFrame to extract and clean interest values from clauses.

    Iterates through the DataFrame to extract interest percentages from 'Sub-Clause' and
    'Matched Clause' columns using check_and_extract_interest, then cleans them using
    postprocess_interest_value. Adds the results as new columns.

    Args:
        input_df (pandas.DataFrame): DataFrame with 'Sub-Clause', 'Matched Clause', and
            'Similarity Score' columns.

    Returns:
        pandas.DataFrame: The input DataFrame with two new columns:
            'Interest Value in Sub-Clause' and 'Interest Value in Matched Clausee'.
    """
    interest_values_sub_clause = []
    interest_values_matched_clause = []
    columns_to_keep = ['Sub-Clause', 'Matched Clause', 'Similarity Score']
    df_col = input_df[columns_to_keep]        
    for row_index in range(len(input_df)):
        new_df_interest = pd.DataFrame(df_col.iloc[row_index]).transpose()
        column1 = 'Sub-Clause'
        column2 = 'Matched Clause'
        text1 = str(new_df_interest[column1])
        text2 = str(new_df_interest[column2])

        interest_value_text1 = check_and_extract_interest(text1)
        print("Interest Value in Text1:", interest_value_text1)
        
        if interest_value_text1:
            try:
                interest_value_text1_postprocess = postprocess_interest_value(interest_value_text1)
                interest_values_sub_clause.append(interest_value_text1_postprocess)
            except:
                interest_values_sub_clause.append('NIL') 
        else:
            interest_values_sub_clause.append('NIL')

        interest_value_text2 = check_and_extract_interest(text2)
        print("Interest Value in Text2 :", interest_value_text2)
        if interest_value_text2:
            try:
                interest_value_text2_postprocess = postprocess_interest_value(interest_value_text2)
                interest_values_matched_clause.append(interest_value_text2_postprocess)
            except:
                interest_values_matched_clause.append('NIL')
        else:
            interest_values_matched_clause.append('NIL')

    input_df['Interest Value in Sub-Clause'] = interest_values_sub_clause
    input_df['Interest Value in Matched Clausee'] = interest_values_matched_clause

    return input_df

def generate_csv_output(query_sentences, passages, pdf_file, output_directory):
    """Generates a CSV file with matched clauses and interest values from PDF passages.

    Compares predefined query sentences against passages using a sentence transformer model
    to find semantic matches. Extracts interest values from matched clauses and saves results
    to a CSV file.

    Args:
        query_sentences (list): List of known sentences to be matched against the sentences in passages.
        passages (list): List of text passages extracted from a PDF.
        pdf_file (str): Path to the input PDF file, used to name the output CSV.
        output_directory (str): Directory where the output CSV will be saved.

    Returns:
        tuple: A tuple containing:
            - str: Path to the generated CSV file.
            - pandas.DataFrame: DataFrame with matched clauses and interest values.

    Raises:
        ValueError: If query_sentences or passages are empty.
    """
    pdf_filename = os.path.splitext(os.path.basename(pdf_file))[0]
    model = SentenceTransformer('all-mpnet-base-v2')
    threshold = 0.6

    #query_sentences = [
    #    "1. Mentioning LC Number(No) on commercial invoice and BOE(Bill of Exchange) is sufficient.",
    #    "2-Interest / Discounting Charges  @ 8.50%  for …. days plus document charges for Rs.2000/- per BOE and Stamp Duty (if applicable) will be charged along with statutory levies i.e GST (@ rate applicable) & TCS @ 0.1% (if applicable) through a Debit Note / Invoice to be attached along with the same Bill of Exchange (BoE) to be raised under this LC. .",
    #    "3. All Discrepancies in the Bills of Exchange to be to be raised under this LC or accompanying documents has been accepted both by LC Opening Bank & Applicants for negotiation and final payment on due date.",
    #    "4. Notwithstanding any provisions or clauses contained in this LC , pre-acceptance of any documents by LC Opening Bank or Applicants is not required for negotiation and final payment on due date.",
    #    "5. On due date payment will be effected as per instructions of negotiating bank.",
    #    "6. Any documents showing description of goods which is commonly accepted in interchangeable terms in steel industry for description of goods in Lc are acceptable",
    #    "7. Any advance received can be deducted from the total value of the invoice to be discounted against this credit.",
    #    "8. This LC is without recourse to drawer.",
    #    "9. As place of expiry is at the counters of negotiating bank, it is sufficient to present the    Bill of Exchange and its accompanying documents to negotiating bank before date of LC expiry and  the  documents  will not  be  refused  on the  ground that  the  documents  received after  the date of  expiry .",
    #    "10. Any Purchase order(PO) number mentioned on Invoices are acceptable",
    #    "11. In case of dispatch to any third party (Ship to), LC to specify the name and address of Ship to Party ",
    #    "12. Only for information material as per Purchase Order no ……. (Or -Sales Contract no ……..  )",
    #    "13. IF BoE raised under this LC is couriered by Negotiating Bank to LC opening bank and gets delayed due to postal delay in this CORONA period, LC opening bank agrees to convey their acceptance of the BoE raised under UCPDC 600, based on soft copy of the BoE sent by negotiating bank through their official mail followed by SFMS message with details mentioned below: 1. Mail ID from which mail sent. 2. Mail l  dated 3.Subject of mail 4. LC  number and amount 5. LC Beneficiary 6.LC Applicant 7.BOE raised amount"
    #]

    if not query_sentences or not passages:
        return "Query or passage sentences are empty. Please provide valid input."

    results = []
    for query_sentence in query_sentences:
        query_embedding = model.encode(query_sentence)
        highest_similarity_score = -1
        most_matched_sentence = 'missing'
        for passage_sentence in passages:
            passage_embedding = model.encode(passage_sentence)
            similarity_score = util.dot_score(query_embedding, passage_embedding)
            if similarity_score > highest_similarity_score:
                highest_similarity_score = similarity_score.item()
                most_matched_sentence = passage_sentence
        if highest_similarity_score >= threshold:
            result_label = most_matched_sentence
        else:
            result_label = 'missing'
            highest_similarity_score = 0
        result = {
            'Sub-Clause': query_sentence,
            'Matched Clause': result_label,
            'Similarity Score': round(highest_similarity_score, 2)
        }
        results.append(result)

    df = pd.DataFrame(results, columns=['Sub-Clause', 'Matched Clause', 'Similarity Score'])
    new_df = df.copy()
    new_df_interest = process_interest_values(new_df)
    
    output_csv = os.path.join(output_directory, f'LC_{pdf_filename}_Matching_Output.csv')
    new_df_interest.to_csv(output_csv, index=False, encoding='utf-8')
    print("New DF Interested", new_df_interest)
    print(f"Extracted Data from LC saved in '{output_csv}'")
    return output_csv, new_df_interest
