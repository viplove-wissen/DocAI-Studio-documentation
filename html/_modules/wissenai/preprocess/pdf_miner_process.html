def transform_bbox(bbox, page_height):
    """
    Transforms the bounding box coordinates from bottom-left origin to top-left origin.
    bbox[0], bbox[1] represent the bottom-left corner
    bbox[2], bbox[3] represent the top-right corner
    After transformation:
    bbox[0], bbox[1] represent the top-left corner
    bbox[2], bbox[3] represent the bottom-right corner
    """
    return (
        bbox[0],
        page_height - bbox[3],
        bbox[2],
        page_height - bbox[1]
    )

def extract_text_with_bbox(file_path,pdf_name):
    """
    Args:
        file_path (str): Path to the input PDF file.
        pdf_name (str): Name or identifier of the PDF file, stored for reference in each text block.

    Returns:
        dict: A dictionary mapping page numbers (int) to a list of dictionaries,
              each containing:
                  - "text_block_id" (int): ID of the text line within the element.
                  - "element_id" (int): ID of the text element (LTTextBoxHorizontal).
                  - "text" (str): The extracted text string.
                  - "pdf_name" (str): The provided PDF name/identifier.
                  - "bbox" (dict): Bounding box with keys "x0", "y0", "x1", "y1" transformed to top-left origin.
                  - "page_number" (int): Page number (1-indexed).
                  - "page_width" (float): Width of the page in points.
                  - "page_height" (float): Height of the page in points.
    """
    laparams = LAParams()
    resource_manager = PDFResourceManager()
    device = PDFPageAggregator(resource_manager, laparams=laparams)
    interpreter = PDFPageInterpreter(resource_manager, device)

    results = {}
    pdf_file_name = get_file_name_without_extension(file_path)

    with open(file_path, 'rb') as fp:
        for page_number, page in enumerate(PDFPage.get_pages(fp)):
            interpreter.process_page(page)
            layout = device.get_result()

            # Extract page width and height
            page_width = layout.bbox[2]
            page_height = layout.bbox[3]

            #'''
            #bbox = text_line.bbox
            #(x0, y0, x1, y1) = bbox
            #x0: The x-coordinate of the bottom-left corner.
            #y0: The y-coordinate of the bottom-left corner.
            #x1: The x-coordinate of the top-right corner.
            #y1: The y-coordinate of the top-right corner.
            #'''
            if page_number + 1 not in results:
                results[page_number + 1] = []
                

            for element_id,element in enumerate(layout):
                if isinstance(element, LTTextBoxHorizontal):
                    for text_line_id,text_line in enumerate(element):
                        bbox = text_line.bbox
                        transformed_bbox = transform_bbox(bbox, page_height)
                        results[page_number + 1].append({ 

                                            "text_block_id" : text_line_id ,
                                            "element_id" : element_id,
                                            "text": text_line.get_text().strip(),
                                            # "bbox": {
                                            #     "x0": bbox[0],
                                            #     "y0": bbox[1],
                                            #     "x1": bbox[2],
                                            #     "y1": bbox[3]
                                            # },
                                            "pdf_name": pdf_name,
                                            "bbox":transformed_bbox,
                                            "page_number": page_number + 1,
                                            "page_width": page_width,
                                            "page_height": page_height
                                            
                                        })

    return results
