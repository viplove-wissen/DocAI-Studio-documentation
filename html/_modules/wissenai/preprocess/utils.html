def enhance_dpi(image=None, image_path=None, new_dpi=300, old_dpi=150):
    """Enhance the DPI of the image.
    Args:
            image (byte_stream): binary byte stream of an image
            image_path(str): Path to the image for inferencing
            new_dpi(int) : what is the new dpi that you want to enhance to
        Returns:
            PIL image object(PIL)
    """
    if image_path:
        image = load_image(image_path)
    scaling_factor = new_dpi / old_dpi

    if isinstance(image, PILImage.Image):
        new_size = (int(image.width * scaling_factor), int(image.height * scaling_factor))
        resized_image = image.resize(new_size, PILImage.LANCZOS)
        return resized_image

    elif isinstance(image, np.ndarray):
        height, width = image.shape[:2]
        new_size = (int(width * scaling_factor), int(height * scaling_factor))
        resized_image = cv2.resize(image, new_size, interpolation=cv2.INTER_LANCZOS4)
        return resized_image

    else:
        raise TypeError("Unsupported image type. Expected PIL.Image.Image or numpy.ndarray.")

def convert_to_binary(image=None, image_path=None):
    """Convert image to binary format using thresholding.
    Args:
            image (byte_stream): binary byte stream of an image
            image_path(str): Path to the image for inferencing
        Returns:
            PIL image object(PIL)
    """
    if image_path:
        image = load_image(image_path)
    if isinstance(image, PILImage.Image):
        gray_image = ImageOps.grayscale(image)
        binary_image = gray_image.point(lambda x: 0 if x < 128 else 255, '1')
        return binary_image
    elif isinstance(image, np.ndarray):
        gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        _, binary_image = cv2.threshold(gray_image, 128, 255, cv2.THRESH_BINARY)
        return binary_image
    else:
        raise TypeError("Unsupported image type. Expected PIL.Image.Image or numpy.ndarray.")

def adaptive_threshold(image=None, image_path=None):
    """Apply adaptive thresholding to the image.
    Args:
            image (byte_stream): binary byte stream of an image
            image_path(str): Path to the image for inferencing
    Returns:
            PIL image object(PIL)
    """
    if image_path:
        image = load_image(image_path)
    if isinstance(image, PILImage.Image):
        gray_image = ImageOps.grayscale(image)
        numpy_image = np.array(gray_image)
        threshold_image = cv2.adaptiveThreshold(numpy_image, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, 11,
                                                2)
        return PILImage.fromarray(threshold_image)
    elif isinstance(image, np.ndarray):
        if len(image.shape) == 3:
            gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        else:
            gray_image = image
        threshold_image = cv2.adaptiveThreshold(gray_image, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, 11,
                                                2)
        return threshold_image
    else:
        raise TypeError("Unsupported image type. Expected PIL.Image.Image or numpy.ndarray.")

def noise_removal(image=None, image_path=None):
    """Remove noise from the image.
    Args:
            image (byte_stream): binary byte stream of an image
            image_path(str): Path to the image for inferencing
    Returns:
            PIL image object(PIL)
    """
    if image_path:
        image = load_image(image_path)

        numpy_image = np.array(image)

    else:
        raise TypeError("Unsupported image type. Expected PIL.Image.Image or numpy.ndarray.")

    if len(numpy_image.shape) == 2:  # Check if the image is grayscale

    else:
        denoised_image = cv2.fastNlMeansDenoisingColored(numpy_image, None, 30, 7, 21)
    return PILImage.fromarray(denoised_image)

def enhance_contrast(image=None, image_path=None):
    """Enhance the contrast of the image.
    Args:
            image (byte_stream): binary byte stream of an image
            image_path(str): Path to the image for inferencing
    Returns:
            PIL image object(PIL)
    """
    if image_path:
        image = load_image(image_path)
    if isinstance(image, PILImage.Image):
        enhancer = ImageEnhance.Contrast(image)
        return enhancer.enhance(2)
    elif isinstance(image, np.ndarray):
        numpy_image = image
        lab = cv2.cvtColor(numpy_image, cv2.COLOR_BGR2LAB)
        l, a, b = cv2.split(lab)
        clahe = cv2.createCLAHE(clipLimit=3.0, tileGridSize=(8, 8))
        cl = clahe.apply(l)
        limg = cv2.merge((cl, a, b))
        contrast_enhanced_image = cv2.cvtColor(limg, cv2.COLOR_LAB2BGR)
        return contrast_enhanced_image
    else:
        raise TypeError("Unsupported image type. Expected PIL.Image.Image or numpy.ndarray.")

def resize_image(image=None, image_path=None, size=(1024, 1024)):
    """Resize the image to a given size.
    Args:
            image (byte_stream): binary byte stream of an image
            image_path(str): Path to the image for inferencing

            PIL image object(PIL)
    """

        image = load_image(image_path)
    if isinstance(image, PILImage.Image):
        return image.resize(size, PILImage.LANCZOS)
    elif isinstance(image, np.ndarray):
        return cv2.resize(image, size, interpolation=cv2.INTER_LANCZOS4)
    else:
        raise TypeError("Unsupported image type. Expected PIL.Image.Image or numpy.ndarray.")

def remove_border(image=None, image_path=None):
    """Remove borders from the image.
    Args:
            image (byte_stream): binary byte stream of an image
            image_path(str): Path to the image for inferencing
    Returns:
            PIL image object(PIL)"""
    if image_path:
        image = load_image(image_path)
    if isinstance(image, PILImage.Image):
        numpy_image = np.array(image)
    elif isinstance(image, np.ndarray):
        numpy_image = image
    else:
        raise TypeError("Unsupported image type. Expected PIL.Image.Image or numpy.ndarray.")

    if len(numpy_image.shape) == 2:
        gray = numpy_image
    else:
        gray = cv2.cvtColor(numpy_image, cv2.COLOR_BGR2GRAY)

    result = numpy_image.copy()
    thresh = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)[1]

    # Remove vertical lines
    vertical_kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (1, 20))
    remove_vertical = cv2.morphologyEx(thresh, cv2.MORPH_OPEN, vertical_kernel, iterations=2)
    cnts = cv2.findContours(remove_vertical, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)[0]
    for c in cnts:
        cv2.drawContours(result, [c], -1, (255, 255, 255), 5)

    # Remove horizontal lines
    horizontal_kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (40, 1))
    remove_horizontal = cv2.morphologyEx(thresh, cv2.MORPH_OPEN, horizontal_kernel, iterations=2)
    cnts = cv2.findContours(remove_horizontal, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)[0]
    for c in cnts:
        cv2.drawContours(result, [c], -1, (255, 255, 255), 5)

    return PILImage.fromarray(result)

def thinning(image=None, image_path=None):
    """Apply thinning to the image using morphological operations.
    Args:
            image (byte_stream): binary byte stream of an image
            image_path(str): Path to the image for inferencing
    Returns:
            PIL image object(PIL)
    """
    if image_path:
        image = load_image(image_path)
    if isinstance(image, PILImage.Image):
        numpy_image = np.array(image)
    elif isinstance(image, np.ndarray):
        numpy_image = image
    else:
        raise TypeError("Unsupported image type. Expected PIL.Image.Image or numpy.ndarray.")

    if len(numpy_image.shape) == 2:
        gray = numpy_image
    else:
        gray = cv2.cvtColor(numpy_image, cv2.COLOR_BGR2GRAY)

    _, binary_image = cv2.threshold(gray, 127, 255, cv2.THRESH_BINARY)

    # Morphological thinning
    size = np.size(binary_image)
    skeleton = np.zeros(binary_image.shape, np.uint8)

    element = cv2.getStructuringElement(cv2.MORPH_CROSS, (3, 3))
    done = False

    while not done:
        eroded = cv2.erode(binary_image, element)
        temp = cv2.dilate(eroded, element)
        temp = cv2.subtract(binary_image, temp)
        skeleton = cv2.bitwise_or(skeleton, temp)
        binary_image = eroded.copy()

        zeros = size - cv2.countNonZero(binary_image)
        if zeros == size:
            done = True

    return PILImage.fromarray(skeleton)

def custom_preprocessing(image=None, image_path=None):
    """Apply complex preprocessing including denoising, thresholding, and thinning."""
    if image_path:
        image = load_image(image_path)
    if isinstance(image, PILImage.Image):
        numpy_image = np.array(image)
    elif isinstance(image, np.ndarray):
        numpy_image = image
    else:
        raise TypeError("Unsupported image type. Expected PIL.Image.Image or numpy.ndarray.")

    # Convert single-channel image to 3-channel image
    if len(numpy_image.shape) == 2:  # Check if the image is grayscale
        numpy_image = cv2.cvtColor(numpy_image, cv2.COLOR_GRAY2BGR)

    dst = cv2.fastNlMeansDenoisingColored(numpy_image, None, 10, 10, 7, 21)
    Istrech = cv2.convertScaleAbs(dst, alpha=1, beta=0)

    roi_gray = cv2.cvtColor(Istrech, cv2.COLOR_BGR2GRAY)
    blur = cv2.GaussianBlur(roi_gray, (3, 3), 0)
    _, th5 = cv2.threshold(blur, 25, 180, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

    bilateral = cv2.bilateralFilter(th5, 5, 75, 75)
    adaptive_thresh = cv2.adaptiveThreshold(bilateral, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, 11, 2)
    _, Ithres = cv2.threshold(adaptive_thresh, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

    edges = cv2.Canny(Ithres, 100, 200)
    edges = cv2.dilate(edges, None, iterations=1)
    edges = cv2.erode(edges, None, iterations=1)

    # Use morphological operations for thinning as ximgproc is unavailable
    size = np.size(edges)
    skeleton = np.zeros(edges.shape, np.uint8)

    element = cv2.getStructuringElement(cv2.MORPH_CROSS, (3, 3))
    done = False

    while not done:
        eroded = cv2.erode(edges, element)
        temp = cv2.dilate(eroded, element)
        temp = cv2.subtract(edges, temp)
        skeleton = cv2.bitwise_or(skeleton, temp)
        edges = eroded.copy()

        zeros = size - cv2.countNonZero(edges)
        if zeros == size:
            done = True

    return PILImage.fromarray(skeleton)

def preprocessing_pipeline(page_image_path, output_directory):
    """Deskew the image and return all intermediate images in PIL format."""
    file_name = get_file_name_without_extension(page_image_path)

    output_images_path = os.path.join(output_directory, f"{file_name}_output_images")
    os.makedirs(output_images_path, exist_ok=True)
    intermediate_images = {}

    # Load the image using PIL
    original_image = load_image(page_image_path)

    # Deskew Image
    deskewed_image = deskew_image(page_image_path)
    intermediate_images['deskewed_image'] = PILImage.fromarray(deskewed_image) if isinstance(deskewed_image,
                                                                                             np.ndarray) else deskewed_image

    # Enhance DPI
    enhanced_dpi_image = enhance_dpi(deskewed_image)
    intermediate_images['enhanced_dpi_image'] = PILImage.fromarray(enhanced_dpi_image) if isinstance(enhanced_dpi_image,
                                                                                                     np.ndarray) else enhanced_dpi_image

    # Convert to binary
    binary_image = convert_to_binary(enhanced_dpi_image)
    intermediate_images['binary_image'] = PILImage.fromarray(binary_image) if isinstance(binary_image,
                                                                                         np.ndarray) else binary_image

    # Apply adaptive thresholding
    threshold_image = adaptive_threshold(binary_image)
    intermediate_images['adaptive_threshold_image'] = PILImage.fromarray(threshold_image) if isinstance(threshold_image,
                                                                                                        np.ndarray) else threshold_image

    # Remove noise
    noise_removed_image = noise_removal(threshold_image)
    intermediate_images['noise_removed_image'] = PILImage.fromarray(noise_removed_image) if isinstance(
        noise_removed_image, np.ndarray) else noise_removed_image

    # Enhance contrast
    contrast_enhanced_image = enhance_contrast(noise_removed_image)
    intermediate_images['contrast_enhanced_image'] = PILImage.fromarray(contrast_enhanced_image) if isinstance(
        contrast_enhanced_image, np.ndarray) else contrast_enhanced_image

    # Resize image
    resized_image = resize_image(contrast_enhanced_image)
    intermediate_images['resized_image'] = PILImage.fromarray(resized_image) if isinstance(resized_image,
                                                                                           np.ndarray) else resized_image

    # Apply edge detection
    edge_detected_image = edge_detection(resized_image)
    intermediate_images['edge_detection_image'] = PILImage.fromarray(edge_detected_image) if isinstance(
        edge_detected_image, np.ndarray) else edge_detected_image

    # Remove borders
    borderless_image = remove_border(edge_detected_image)
    intermediate_images['remove_border_image'] = PILImage.fromarray(borderless_image) if isinstance(borderless_image,
                                                                                                    np.ndarray) else borderless_image

    # Apply thinning
    thinned_image = thinning(borderless_image)
    intermediate_images['thinned_image'] = PILImage.fromarray(thinned_image) if isinstance(thinned_image,
                                                                                           np.ndarray) else thinned_image

    # Apply skeletonization
    skeletonized_image = skeletonization(thinned_image)
    intermediate_images['skeletonized_image'] = PILImage.fromarray(skeletonized_image) if isinstance(skeletonized_image,
                                                                                                     np.ndarray) else skeletonized_image

    # Apply complex preprocessing
    complex_processed_image = custom_preprocessing(skeletonized_image)
    intermediate_images['custom_processed_image'] = PILImage.fromarray(complex_processed_image) if isinstance(
        complex_processed_image, np.ndarray) else complex_processed_image

    # Save intermediate images
    img_name = os.path.basename(page_image_path)
    for key, img in intermediate_images.items():
        output_directory_path = os.path.join(output_images_path, f"{key}.jpeg")
        img.save(output_directory_path)

    return intermediate_images, file_name, output_images_path

def rotate(
        image: np.ndarray, angle: float, background: Union[int, Tuple[int, int, int]]
) -> np.ndarray:
    old_width, old_height = image.shape[:2]
    angle_radian = math.radians(angle)
    width = abs(np.sin(angle_radian) * old_height) + abs(np.cos(angle_radian) * old_width)
    height = abs(np.sin(angle_radian) * old_width) + abs(np.cos(angle_radian) * old_height)

    image_center = tuple(np.array(image.shape[1::-1]) / 2)
    rot_mat = cv2.getRotationMatrix2D(image_center, angle, 1.0)
    rot_mat[1, 2] += (width - old_width) / 2
    rot_mat[0, 2] += (height - old_height) / 2
    return cv2.warpAffine(image, rot_mat, (int(round(height)), int(round(width))), borderValue=background)

def de_skew(image=None, image_path=None):
    """
    Deskew any image to correct orientaion
    Args:
            image (byte_stream): binary byte stream of an image
            image_path(str): Path to the image for inferencing
    Returns:
            PIL image object(PIL)
    """
    if image_path:
        image = cv2.imread(image_path)
    cv2_image = np.array(image)
    # Convert RGB to BGR (which OpenCV uses)
    image = cv2.cvtColor(cv2_image, cv2.COLOR_RGB2BGR)
    grayscale = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    angle = determine_skew(grayscale)
    rotated = rotate(image, angle, (0, 0, 0))
    return rotated
