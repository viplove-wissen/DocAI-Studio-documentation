def image_to_base64(image):
    buffer = BytesIO()
    image.save(buffer, format='JPEG')
    buffer.seek(0)
    return base64.b64encode(buffer.getvalue()).decode('utf-8')

@app.post("/input_file")
async def do_preprocessing(file: Annotated[UploadFile, Form()], request: Request):
    base_url = str(request.base_url)

    # Clear the lists whenever a new file is uploaded
    global file_name_list, file_path_list, output_json
    file_name_list = []
    file_path_list = []
    output_json = []

    save_path = os.path.join(root_file, file.filename)

    with open(save_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)

    file_path = save_path
    file_path_list.append(file_path)
    file_name = file.filename

    preprocessed_images_dict, file_name, file_preprocessed_images_dir = preprocessing_pipeline(file_path,
                                                                                               output_directory)

    file_name_list.append(file_name)
    file_images_urls = []

    for img, file_name in enumerate(os.listdir(file_preprocessed_images_dir)):
        file_path = os.path.join(file_preprocessed_images_dir, file_name)
        print("file_path", file_path)
        if file_path.endswith((".jpg", ".jpeg", ".png", ".tiff", ".gif")):
            img_url = base_url + 'image/' + str(quote(file_path))
            # print(img_url)
            file_images_urls.append(img_url)

    output_details = []
    preprocessing_steps_key = []

    for key, value in preprocessed_images_dict.items():
        preprocessing_steps_key.append(key)

    for id_no, preprocessed_image_url in enumerate(file_images_urls):
        file_name = get_file_name_without_extension(preprocessed_image_url)

        output_details.append({
            'preprocessed_image_url_block_id': id_no + 1,
            file_name: preprocessed_image_url,
        })

    output_json.append({

        'preprocessed_images_urls': output_details,
        'preprocessing_steps_keys': preprocessing_steps_key

    })

    # print("output_json[0]",output_json[0])

    return output_json[0]

@app.post("/enhance_dpi")
async def do_enchance(file: Annotated[UploadFile, File()], ):

    process_multilingual_cost()
    try:
        file_content = await file.read()
        filename = file.filename
        image = Image.open(io.BytesIO(file_content))
        new_dpi = config['dpi']['new_dpi']

        output_dir = root_path + output_folder

        return StreamingResponse(byte_io, media_type="image/jpeg", status_code=200)
    except:
        Config.logger.exception('An error occurred while handling request for /error')
        return JSONResponse(content="Something went wrong", status_code=400)

@app.post("/to_binary_multi")
async def to_binary_multi_helper(files: Annotated[List[UploadFile], File()]):
    Config.logger.info("Converting to binary started")
    process_multilingual_cost()

    root_path = os.getcwd()
    output_folder = config['paths']['output_directory']
    output_dir = os.path.join(root_path, output_folder)
    os.makedirs(output_dir, exist_ok=True)

    responses = []

    for file in files:
        try:
            file_content = await file.read()
            filename = file.filename
            image = Image.open(io.BytesIO(file_content))

            output_image = convert_to_binary(image)
            if output_image.mode == 'RGBA':
                output_image = output_image.convert('RGB')

            byte_io = BytesIO()
            output_image.save(byte_io, format='JPEG')  # Save image to the byte stream
            byte_io.seek(0)

            output_file_path = os.path.join(output_dir, f"binary_{filename}")
            output_image.save(output_file_path, format='JPEG')

            # Add the StreamingResponse for each file to the responses list
            responses.append({
                "filename": filename,
                "output_path": output_file_path,
                "image": StreamingResponse(byte_io, media_type="image/jpeg")
            })

        except Exception as e:
            Config.logger.exception(f'An error occurred while processing file {file.filename}')
            return JSONResponse(content=f"Something went wrong with file {file.filename}", status_code=400)

    # Return the responses in a list
    return JSONResponse(content=[{"filename": res["filename"], "output_path": res["output_path"]} for res in responses])
