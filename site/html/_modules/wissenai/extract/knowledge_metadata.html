def search_headers_in_json(json_data, query_text, search_method='fuzzy', label_filter=None, fuzzy_threshold=70, embedding_threshold=70):
    """
    Search for headers in JSON data similar to the query text using the specified method,
    optionally filtering by label type.
    
    Args:
        json_data (list): List of JSON objects containing header and content information.
        query_text (str): The query text to match against headers.
        search_method (str): Search method ('fuzzy' or 'embedding').
        label_filter (str, optional): Filter headers by label_name ('Section-header','Table-header' or 'Page-header'). If None, search all headers.
        fuzzy_threshold (int): Minimum fuzzy similarity score for a match (0-100).
        embedding_threshold (int): Minimum embedding similarity score for a match (0-100).
    
    Returns:
        str: Formatted string of matched headers, their content, and similarity scores.
    """
    def load_embedding_model():
        """Load a pre-trained sentence transformer model."""
        return SentenceTransformer('all-MiniLM-L6-v2')

    def compute_fuzzy_similarity(query, text):
        """
        Compute fuzzy similarity score between query and text.
        
        Args:
            query (str): Query text.
            text (str): Text to compare against (header).
        
        Returns:
            float: Fuzzy similarity score (0-100).
        """
        return fuzz.partial_ratio(query.lower(), text.lower())

    def compute_embedding_similarity(query, text, model):
        """
        Compute cosine similarity between query and text embeddings.
        
        Args:
            query (str): Query text.
            text (str): Text to compare against (header).
            model: Pre-trained sentence transformer model.
        
        Returns:
            float: Cosine similarity score (0 to 100).
        """
        query_embedding = model.encode(query, convert_to_tensor=True)
        text_embedding = model.encode(text, convert_to_tensor=True)
        similarity = util.cos_sim(query_embedding, text_embedding).item()
        return similarity * 100

    def format_results(results):
        """
        Format the search results into a readable string.
        
        Args:
            results (list): List of matched headers and their content.
        
        Returns:
            str: Formatted string of results.
        """
        if not results:
            return "No headers found matching the query."
        
        output = []
        for result in results:
            output.append(f"Header: {result['header']}")
            output.append(f"Label: {result['label_name']}")
            output.append(f"{result['score_type'].capitalize()} Similarity Score: {result['score']:.2f}%")
            output.append(f"Content:\n{result['content']}\n")
        
        return "\n".join(output)

    # Validate search method
    if search_method not in ['fuzzy', 'embedding']:
        return "Invalid search method. Use 'fuzzy' or 'embedding'."

    # Initialize results list
    results = []
    
    # Load embedding model if needed
    if search_method == 'embedding':
        model = load_embedding_model()
    
    # Process each JSON item
    for item in json_data:
        header = item.get('header', '')
        content = item.get('content', '')
        label_name = item.get('label_name', '')
        
        # Apply label filter if specified
        if label_filter and label_name != label_filter:
            continue
        
        if search_method == 'fuzzy':
            score = compute_fuzzy_similarity(query_text, header)
            if score >= fuzzy_threshold:
                results.append({
                    'header': header,
                    'content': content,
                    'label_name': label_name,
                    'score': score,
                    'score_type': 'fuzzy'
                })
        elif search_method == 'embedding':
            score = compute_embedding_similarity(query_text, header, model)
            if score >= embedding_threshold:
                results.append({
                    'header': header,
                    'content': content,
                    'label_name': label_name,
                    'score': score,
                    'score_type': 'embedding'
                })
    
    # Sort results by score in descending order
    results.sort(key=lambda x: x['score'], reverse=True)
    
    # Format and return results
    return format_results(results)
