def userquery(request_params):
    """
    Handles a user query by extracting the answer from a document image using a visual question answering (VQA) model.

    Args:
        request_params (dict): Dictionary containing parameters for the request, including:
            - 'basePath' (str): The base directory for documents.
            - 'tenantId' (str): Tenant identifier.
            - 'documentId' (str): Document identifier.
            - 'templateId' (str): Template identifier.
            - 'UserQuery' (str): The user's query string.

    Returns:
        dict: A dictionary containing the original user query and the VQA response.
    """
    base_path = request_params['basePath']
    print(base_path)
    tenantId = request_params['tenantId']
    documentId = request_params['documentId']
    templateId = request_params['templateId']
    question = request_params["UserQuery"]
    _path = os.path.join(base_path, tenantId, templateId, documentId)
    image = Image.open(_path)
    answer = chat(_path, question)
    # answer = answer['answer']
    output = {'userquery': question, 'response': answer}
    return output

def multivqa(image_path = None, question = None):
    """Function that performs inferencing of images for visual question answering, works best on PID, KYC type of documents of any strcture.

            image_path (str): Path to the image to be used for inferencing
            question(str) : question to be asked from the document

        Returns:
            <class 'dict'>
            example of a output format is : {'userquery': 'what is the name?', 'response': 'Aarti Mehta'}

        """
    answer = chat(image_path, question)
    output = {'userquery': question, 'response': answer}
    return output

def chat(_path, question):
    """
    Sends a single image and question to the document intelligence model for inference.

    Args:
        _path (str): Path to the image file.
        question (str): The question to ask about the image.

    Returns:
        str: The answer extracted by the model.
    """
    output = docprompt([{"doc": _path, "prompt": [question]}])
    print(output)
    temp_dict = output[0]["result"][0]
    answer = temp_dict["value"]
    print(answer)
    return answer

def extract(path_, questions):
    """
    Extracts answers for a set of questions from a single image using the document intelligence model.

    Args:
        path_ (str): Path to the image file.
        questions (dict): Dictionary of questions where values are question strings.

    Returns:
        list: A list of answers corresponding to the provided questions.
    """
    ans = []
    for question in questions.values():
        print(question)

        output = docprompt([{"doc": path_, "prompt": [question]}])
        temp_dict = output[0]["result"][0]
        answer = temp_dict["value"]
        print(answer)
        ans.append(answer)
    return ans
