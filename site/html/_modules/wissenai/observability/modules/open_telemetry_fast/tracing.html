def configure_tracing(app: FastAPI, _SERVICE_NAME: str, OTEL_EXPORTER_OTLP_ENDPOINT: str):
    """
    Configure OpenTelemetry tracing for a FastAPI application.

    Args:
        app (FastAPI): The FastAPI application instance to instrument.
        _SERVICE_NAME (str): The name of the service for which the traces are being generated.
        OTEL_EXPORTER_OTLP_ENDPOINT (str): The endpoint to which the OTLP spans should be exported.

    - Sets up a TracerProvider with the specified service name.
    - Configures an OTLPSpanExporter that sends trace data to the specified endpoint.
    - Adds a BatchSpanProcessor to handle exporting spans in batches.
    - Instruments the FastAPI application to automatically generate trace spans for each request.
    """
    # Set up OpenTelemetry Tracer
    trace.set_tracer_provider(
        TracerProvider(resource=Resource.create({SERVICE_NAME: _SERVICE_NAME}))
    )
    
    # Set up the OTLP exporter
    otlp_exporter = OTLPSpanExporter(endpoint=OTEL_EXPORTER_OTLP_ENDPOINT, insecure=True)

    # Add the exporter to the TracerProvider
    span_processor = BatchSpanProcessor(otlp_exporter)
    trace.get_tracer_provider().add_span_processor(span_processor)

    # Instrument FastAPI with OpenTelemetry
    FastAPIInstrumentor.instrument_app(app)

def traced(span_name):
    """
    A decorator for manually creating and managing trace spans.

    Args:
        span_name (str): The name of the span to be created.

    Returns:
        decorator: A function decorator that wraps the target function with tracing logic.

    - Creates a new span with the specified name whenever the decorated function is called.
    - Ensures the span is active for the duration of the function execution.
    - The function must be asynchronous (`async def`) as the wrapper is designed for async functions.
    """
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            with tracer.start_as_current_span(span_name):
                return await func(*args, **kwargs)
        return wrapper
    return decorator
