def configure_metrics(_SERVICE_NAME: str, OTEL_EXPORTER_OTLP_ENDPOINT: str):
    """
    Configures the OpenTelemetry metrics setup for the application.

    Args:
        _SERVICE_NAME (str): The name of the service for which metrics are collected.
        OTEL_EXPORTER_OTLP_ENDPOINT (str): The endpoint URL for the OTLP metric exporter.
    """
    # Exporter to export metrics to Prometheus
    resource=Resource.create({SERVICE_NAME: _SERVICE_NAME})
    
    exporter = OTLPMetricExporter(endpoint=OTEL_EXPORTER_OTLP_ENDPOINT, insecure=True)
    reader = PeriodicExportingMetricReader(exporter)
    
    # reader = PrometheusMetricReader()
    meter_provider = MeterProvider(resource=resource, metric_readers=[reader])
    # Set the global MeterProvider to the one configured
    set_meter_provider(meter_provider)
